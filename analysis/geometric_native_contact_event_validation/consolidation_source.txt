def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def read(path):
    return json.loads((ROOT / path).read_text())

def write(path, value):
    target = OUT / path
    if target.exists():
        raise RuntimeError(f'Preserve existing evidence: {target}')
    target.write_text(json.dumps(value, sort_keys=True, indent=2, allow_nan=False)+'\n')

def seal_valid(raw, key):
    body = {k:v for k,v in raw.items() if k != key}
    assert hashlib.sha256(encoded(body)).hexdigest() == raw[key], key

def manifest_audit(path):
    saved = read(path)
    changed = [name for name, expected in saved.items() if sha(ROOT/name) != expected]
    return dict(path=path, sha256=sha(ROOT/path), path_count=len(saved), changed_paths=changed)

def scalar_interval(raw):
    return (Fraction(raw['lower']['numerator'],raw['lower']['denominator']),
            Fraction(raw['upper']['numerator'],raw['upper']['denominator']))
