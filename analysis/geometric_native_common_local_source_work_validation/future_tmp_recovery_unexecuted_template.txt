def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
        ensure_ascii=False, allow_nan=False).encode()


def value_digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()


def require(condition, label):
    if not condition:
        raise ValueError(label)


def exact(actual, expected, label):
    require(type(actual) is type(expected) and actual == expected, label)
    if type(actual) in (dict, list, tuple):
        require(encoded(actual) == encoded(expected), label + ': complete literal encoding')


def valid_pin(pin):
    return type(pin) is str and len(pin) == 64 and all(c in '0123456789abcdef' for c in pin)


def regular(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path and path.is_file()
            and not path.is_symlink(), 'Canonical regular permanent source: ' + str(path))
    return path


def raw(path):
    result = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            result.update(block)
    return result.hexdigest()


def sealed(path, key, *, raw_pin=None, selfseal=None):
    path = regular(path)
    if raw_pin is not None:
        require(valid_pin(raw_pin), 'Literal raw pin: ' + str(path))
        exact(raw(path), raw_pin, 'Trusted saved raw bytes: ' + str(path))
    value = json.loads(path.read_bytes())
    require(type(value) is dict and key in value, 'Complete saved sealed object: ' + str(path))
    body = dict(value)
    actual = body.pop(key)
    exact(actual, value_digest(body), 'Complete saved selfseal: ' + str(path))
    if selfseal is not None:
        exact(actual, selfseal, 'Saved selfseal association: ' + str(path))
    return value


def console_json(path):
    regular(path)
    lines = Path(path).read_text().splitlines()
    return json.loads(next(line for line in reversed(lines) if line.startswith('{')))
