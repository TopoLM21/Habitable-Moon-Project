def f(v):
 assert type(v) is list and len(v)==2 and all(type(x) is int for x in v)
 assert v[1]>0;q=F(*v);assert v==[q.numerator,q.denominator];return q

def b(v,c):
 assert type(v) is dict and set(v)=={'lo','hi'}
 return c['I'](f(v['lo']),f(v['hi']))
def vbound(v,c):return tuple(b(x,c) for x in v)
def mbound(v,c):return tuple(vbound(row,c) for row in v)
def rnd(x,c):
 x=x if hasattr(x,'lo') else c['I'](x)
 return c['I'](*outward_relative_grid((x.lo,x.hi),256))
def dot(a,b):return sum((x*y for x,y in zip(a,b)),F(0))
def add(a,b):return tuple(x+y for x,y in zip(a,b))
def sub(a,b):return tuple(x-y for x,y in zip(a,b))
def neg(a):return tuple(-x for x in a)
def sc(a,t):return tuple(x*t for x in a)
def cross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])
def mv(M,v):return tuple(dot(row,v) for row in M)
def sym2(a,b,v):return sc(add(cross(a,cross(b,v)),cross(b,cross(a,v))),F(1,2))
def unitaxis(name):return tuple(F(int(k==name)) for k in 'xyz')
def square(v,c):return c['I'](F(0) if v.lo<=0<=v.hi else min(v.lo*v.lo,v.hi*v.hi),max(v.lo*v.lo,v.hi*v.hi))
def sqrtcap(q,precision=256):
 assert q>=0;scale=10**precision;n=math.isqrt((q*scale*scale).numerator//(q*scale*scale).denominator)
 if F(n,scale)**2<q:n+=1
 assert F(n,scale)**2>=q and (n==0 or F(n-1,scale)**2<q)
 return F(n,scale)

def scalar_twojets(subside,overside,frame,coordinate_order,subowner,overowner,c):
 """Independent receiver cancellation h=e0.yo-(Ro e0).(Rs ys).

    The receiver point/director rotation cancels before interval arithmetic.
    Thus owner/over-body and owner/over-radial mixed entries are EXACT zero.
    """
 assert subowner!=overowner
 I=c['I'];n=len(coordinate_order);ys=subside['y'];B=[I(0)]*n;H=[[I(0) for _ in range(n)] for _ in range(n)]
 eta=rnd(dot(frame,sub(overside['delta'],subside['delta'])),c)
 owners={owner:{name:i for i,co in enumerate(coordinate_order) if co[0]=='owner' and co[1]==owner for name in (co[2],)} for owner in (subowner,overowner)}
 for owner in (subowner,overowner):assert set(owners[owner])==set('xyz')
 for i,col in subside['first'].items():B[i]=rnd(B[i]-dot(frame,col),c)
 for i,col in overside['first'].items():B[i]=rnd(B[i]+dot(frame,col),c)
 for (i,j),value in subside['mixed'].items():H[i][j]=H[j][i]=rnd(H[i][j]-dot(frame,value),c)
 for (i,j),value in overside['mixed'].items():H[i][j]=H[j][i]=rnd(H[i][j]+dot(frame,value),c)
 for name,idx in owners[subowner].items():B[idx]=rnd(-dot(unitaxis(name),cross(ys,frame)),c)
 for name,idx in owners[overowner].items():B[idx]=rnd(dot(unitaxis(name),cross(ys,frame)),c)
 for owner in (subowner,overowner):
  for na,i in owners[owner].items():
   for nb,j in owners[owner].items():
    H[i][j]=rnd(-dot(frame,sym2(unitaxis(na),unitaxis(nb),ys)) if owner==subowner
                else -dot(sym2(unitaxis(na),unitaxis(nb),frame),ys),c)
 for ns,i in owners[subowner].items():
  for no,j in owners[overowner].items():H[i][j]=H[j][i]=rnd(-dot(cross(unitaxis(no),frame),cross(unitaxis(ns),ys)),c)
 for index,col in subside['first'].items():
  for name,i in owners[subowner].items():H[i][index]=H[index][i]=rnd(-dot(frame,cross(unitaxis(name),col)),c)
  for name,i in owners[overowner].items():H[i][index]=H[index][i]=rnd(-dot(cross(unitaxis(name),frame),col),c)
 return eta,tuple(B),tuple(tuple(row) for row in H)

def fixed_owner_atom_path(source,subside,overside,body_steps,radial_steps,branch,c):
 """Independent polynomial path: a(t)=a+t da,r(t)=r+t dr.
    Eta is quadratic; traction*eta' is cubic. No stored historical work changes.
    """
 I=c['I'];coeff=[]
 for side in (subside,overside):
  da=tuple(body_steps.get(i,F(0)) for i in side['body_indices']);dr=radial_steps.get(side['radial_index'],F(0))
  du=tuple(sum((x*psi[j] for x,psi in zip(da,side['modes'])),I(0)) for j in range(3))
  dG=tuple(tuple(sum((x*g[i][j] for x,g in zip(da,side['G'])),I(0)) for j in range(3)) for i in range(3))
  c1=add(add(du,mv(dG,side['L'])),sc(mv(side['F'],side['v']),dr))
  c2=sc(mv(dG,side['v']),dr);coeff.append((side['delta'],c1,c2))
 frames=tuple(vbound(e,c) for e in source['archival_old_reference']['receiver_director_bounds']);A=f(source['original_area_m2'])
 t=tuple(f(x) for x in source['literal_source_tractions_pa']);K=tuple(f(x) for x in source['series_stiffness_exact_pa_m'])
 if branch=='relaxed_zero_strength_shear':K=(K[0],F(0));assert t[1]==0
 out=[]
 for channel,(e,t0,k) in enumerate(zip(frames,t,K)):
  eta=tuple(dot(e,sub(coeff[1][i],coeff[0][i])) for i in range(3));c0,c1,c2=eta;tref=I(t0)+k*c0
  bernstein=(c0,c0+c1/2,c0+c1+c2)
  interval_range=I(min(x.lo for x in bernstein),max(x.hi for x in bernstein))
  traction_range=I(t0)+k*interval_range
  if channel==0:assert traction_range.hi<0,'WHOLE joint path stays on compressed normal branch'
  power=(tref*c1,2*tref*c2+k*c1*c1,3*k*c1*c2,2*k*c2*c2)
  integral=rnd(A*sum((x/F(i+1) for i,x in enumerate(power)),I(0)),c)
  increment=rnd(A*(tref*(c1+c2)+k*square(c1+c2,c)/2),c)
  out.append(dict(eta_polynomial=eta,power_polynomial=power,work_integral=integral,potential_difference=increment,
                  scalar_range_bound=interval_range,traction_range_bound=traction_range,
                  power_convention='PLUS_POTENTIAL_DIFFERENTIAL_DU',
                  equality_is_correlated_algebra=True))
 # Physical point/attachment power uses the independently expanded vector
 # force against both complete quadratic driver velocities, hence -dU/dt.
 forcepoly=tuple(tuple(A*sum(((I(t[a])+K[a]*out[a]['eta_polynomial'][0]) if degree==0 else K[a]*out[a]['eta_polynomial'][degree])*frames[a][j] for a in range(2)) for j in range(3)) for degree in range(3))

 physical=(dot(forcepoly[0],velocity0),dot(forcepoly[1],velocity0)+dot(forcepoly[0],velocity1),dot(forcepoly[2],velocity0)+dot(forcepoly[1],velocity1),dot(forcepoly[2],velocity1))
 plus=tuple(A*sum((channel['power_polynomial'][i] for channel in out),I(0)) for i in range(4))
 for actual,potential in zip(physical,plus):
  residual=actual+potential;assert residual.lo<=0<=residual.hi,'whole polynomial physical power is MINUS potential power'
 physical_integral=rnd(sum((coefficient/F(i+1) for i,coefficient in enumerate(physical)),I(0)),c)
 for channel in out:
  channel['complete_physical_attachment_power_polynomial']=physical
  channel['complete_physical_work_integral']=physical_integral
  channel['physical_power_convention']='MINUS_POTENTIAL_DIFFERENTIAL_DU'
 return tuple(out)

class Interval:
 """Exact rational interval arithmetic; no floating point or midpoint."""
 __slots__=('lo','hi')
 def __init__(self,lo,hi=None):
  self.lo=F(lo);self.hi=self.lo if hi is None else F(hi);assert self.lo<=self.hi

 __radd__=__add__
 def __neg__(self):return Interval(-self.hi,-self.lo)
 def __sub__(self,o):return self+-asI(o)
 def __rsub__(self,o):return asI(o)+-self
 def __mul__(self,o):
  o=asI(o);p=(self.lo*o.lo,self.lo*o.hi,self.hi*o.lo,self.hi*o.hi);return Interval(min(p),max(p))
