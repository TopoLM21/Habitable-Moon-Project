def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
        ensure_ascii=False, allow_nan=False).encode()

def digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()

def read_authority(path, pin):
    raw = path.read_bytes()
    if hashlib.sha256(raw).hexdigest() != pin:
        raise ValueError('The immutable complete authority changed: '+str(path))
    return json.loads(raw)

    key += (hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),)

    started = time.monotonic()
    hot903, cold899, work891, transport879 = saved.source_packets()
    source895 = saved.geometry_packets()[0]
    cold923 = read_authority(AUTHORITY923, PIN923)
    current907 = read_authority(AUTHORITY907, PIN907)
    source_read_seconds = time.monotonic()-started

    cold_carrier = saved.cold_carrier()

def run(output):
    from tectonics import geometric_native_common_preparation_contacts as contact_api
    from tectonics import geometric_native_common_preparation_balance as balance_api
    started = time.monotonic()
    ctx = context()
    contacts, balance = ctx['contacts_system'], ctx['balance_system']
    contacts_dto = contact_api.native_common_preparation_contacts_to_dict(contacts)
    balance_dto = balance_api.native_common_preparation_balance_to_dict(balance)
    contacts_restart = contact_api.native_common_preparation_contacts_from_dict(
        contacts_dto, ctx['source907'], ctx['source891'], ctx['preparations'],
        source895=ctx['source895'], provenance=contacts.provenance)
    balance_restart = balance_api.native_common_preparation_balance_from_dict(
        balance_dto, ctx['cold923'], ctx['cold899'], ctx['hot903'], ctx['current907'],
        contacts_system=contacts_restart)
    contact_equal = contact_api.native_common_preparation_contacts_to_dict(contacts_restart) == contacts_dto
    balance_equal = balance_api.native_common_preparation_balance_to_dict(balance_restart) == balance_dto
    assert contact_equal and balance_equal
    zero = (F(0),)*8
    internal = tuple(F((-1 if i%2 else 1)*(i+1), 1000000) for i in range(8))
    specs = [('lambda_'+str(lam), lam, zero) for lam in
        (F(0),F(1,1000),F(1,100),F(1,4),F(1,2),F(3,4),F(1))]
    specs += [('half_internal', F(1,2), internal), ('endpoint_internal', F(1), internal)]
    queries = []
    query_started = time.monotonic()
    for label, lam, increments in specs:
        c = contact_api.evaluate_native_common_preparation_contacts(contacts,
            preparation_fraction=lam, internal_coordinates=increments, precision=96)
        b = balance_api.evaluate_native_common_preparation_balance(balance,
            preparation_fraction=lam, internal_coordinates=increments, precision=96)
        cdto = contact_api.native_common_preparation_contacts_response_to_dict(c)
        bdto = balance_api.native_common_preparation_balance_response_to_dict(b)
        cr = contact_api.native_common_preparation_contacts_response_from_dict(cdto,
            system=contacts_restart, expected_preparation_fraction=lam,
            expected_internal_coordinates=increments, precision=96)
        br = balance_api.native_common_preparation_balance_response_from_dict(bdto,
            system=balance_restart, expected_preparation_fraction=lam,
            expected_internal_coordinates=increments, precision=96)
        assert contact_api.native_common_preparation_contacts_response_to_dict(cr) == cdto
        assert balance_api.native_common_preparation_balance_response_to_dict(br) == bdto
        counts = {}
        for atom in map(json.loads, c.atoms_json):
            branch = atom['loading']['branch']
            counts[branch] = counts.get(branch,0)+1
        assert len(b.gradient_bounds)==15 and len(b.hessian_bounds)==15
        assert all(len(row)==15 for row in b.hessian_bounds)
        assert b.common_owner_rotation_gradient_bounds_nm[2].hi < 0
        assert not b.general_state_mixed_operator_certified
        queries.append(dict(label=label, preparation_fraction=[lam.numerator,lam.denominator],
            internal_coordinates=[[v.numerator,v.denominator] for v in increments],
            contact_response=cdto, balance_response=bdto, branch_counts=counts,
            complete_contacts_response_restart_equal=True,
            complete_balance_response_restart_equal=True))
    assert queries[0]['branch_counts'] == {'intact_or_compressed':16}
    assert queries[2]['branch_counts'].keys() == {'intact_or_compressed','monotone_softening','fully_failed_open'}
    assert queries[6]['branch_counts'] == {'fully_failed_open':12,'intact_or_compressed':4}
    assert balance_api._bounds(balance_api._saved(queries[6]['balance_response'][
        'independent_owner_relaxed_potential_work_lower_bound_j'])).lo > 0
    differences=[]
    for before,after in ((F(0),F(1)),(F(1),F(0)),(F(1,2),F(1,2))):
        difference = balance_api.compare_native_common_preparation_balance_states(balance,before,after)
        assert balance_api.native_common_preparation_balance_state_difference_from_dict(
            difference,system=balance_restart,before_fraction=before,after_fraction=after)==difference
        assert not difference['temporal_path_work_certified']
        differences.append(difference)
    try:
        balance_api.require_native_common_preparation_full_physical_solve(balance)
    except ValueError as error:
        full_solve_rejection=str(error)
    else:
        raise AssertionError('The incomplete physical solve must reject missing body loading')
    assert 'missing86 loads cannot be filled with zero' in full_solve_rejection
    result = dict(format='native-common-joint-preparation-native-1',
        status='PASS_NATIVE927_DECLARED_PREPARATION_CONTACT_LOAD_ACTIVE15',
        contacts_system=contacts_dto,balance_system=balance_dto,queries=queries,
        signed_state_differences=differences,full_physical_solve_rejection=full_solve_rejection,
        complete_contacts_restart_equal=contact_equal,complete_balance_restart_equal=balance_equal,
        joint_preparation_active_block_computed=True,active_mixed_coordinate_count=15,
        full_mixed_coordinate_count=101,missing_body_modal_load_derivative_coordinate_count=86,
        full_mixed_force_and_tangent_certified=False,cold_constitutive_preparation_work_computed=True,
        reference_preparation_work_paid=False,full_system_preparation_work_certified=False,
        historical_archived_three_dimensional_prestress_recovered=False,
        new_preparation_path_constitutive_admissibility_certified=False,
        new_mixed_force_equilibrium_certified=False,whole_clock_all_pair_admission_authorized=False,
        physical_material_admission_authorized=False,full_system_temporal_path_work_certified=False,
        clocks_advanced=False,source_geometry_elapsed_time_s=0,contact_geometry_elapsed_time_s=2000000,
        memory_elapsed_time_s=1000000,material_geometry_elapsed_time_s=1000000,
        accepted_original_native_intervals=1,accepted_new_native_intervals=0,
        original_point_1_completed=False,original_point_1_remaining_work_blocks=1,
        previous_tensor_quadrature_reexecuted=False,historical_force_solve_reexecuted=False,
        remaining_physical_conditions=[
            'Physical material/force/prestress/work handover, full preparation and load work, general-state mixed operator and NEW equilibrium',
            'Whole-clock all-pair admission, including reverse C5 to C1',
            'Accepted continuation with independent temporal path work, balances and restart'],
        timings=dict(ctx['timings'],new_queries_and_restart_seconds=time.monotonic()-query_started,
            total_seconds=time.monotonic()-started))
    result['sha256']=digest(result)
    raw=encoded(result)+b'\n'
    output=Path(output)
    output.parent.mkdir(parents=True,exist_ok=True)
    with output.open('xb') as stream: stream.write(raw)
    sys.stdout.buffer.write(raw)
    return result
