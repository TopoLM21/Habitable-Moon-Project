def encoded(x):
    return json.dumps(x, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()

def digest(x):
    return hashlib.sha256(encoded(x)).hexdigest()

def raw_sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        while block := stream.read(1<<20):
            h.update(block)
    return h.hexdigest()

def require(ok, label):
    if not ok:
        raise AssertionError(label)

def rat(x):
    require(type(x) is not bool, 'A Boolean cannot replace a source rational')
    if type(x) is list:
        require(len(x)==2 and all(type(y) is int for y in x) and x[1]>0, 'Exact source rational pair')
        return F(*x)
    return F.from_float(x) if type(x) is float else F(x)

def num(x):
    x = x if type(x) is F else rat(x)
    return mp.mpf(x.numerator)/x.denominator

def value(x):
    if type(x) is dict and set(x)=={'lo', 'hi'}:
        # A fixed, authenticated enclosure representative is only an algebra
        # diagnostic, never an accepted material point or chronology root.
        return num((rat(x['lo'])+rat(x['hi']))/2)
    return num(x)

def vector(x):
    return tuple(value(v) for v in x)

def matrix(x):
    return tuple(vector(row) for row in x)

def bounds(raw):
    if type(raw) is dict and set(raw)=={'lo', 'hi'}:
        return num(raw['lo']), num(raw['hi'])
    v = num(raw)
    return v, v

def compare(v, raw, label, counts):
    lo, hi = bounds(raw)
    margin = TOL*max(mp.mpf(1), abs(v), abs(lo), abs(hi))
    require(lo-margin <= v <= hi+margin, label+' outside native complete interval')
    counts['scalar_enclosure_checks'] += 1

def compare_tree(v, raw, label, counts):
    if type(v) in (tuple, list):
        require(len(v)==len(raw), label+' complete array shape')
        for i, (entry, receipt) in enumerate(zip(v, raw)):
            compare_tree(entry, receipt, label+'.'+str(i), counts)
    else:
        compare(v, raw, label, counts)

def identity(v, label, counts):
    require(abs(v) <= TOL*max(1,abs(v)), label+' independent identity')
    counts['analytic_identity_checks'] += 1

class Jet:
    """Independent sparse true first/second derivatives in the 15-block.

    Lambda has index14. Six fresh owner coordinates have indices0..5 and
    the eight literal original internal drivers indices6..13.
    """
    __slots__ = ('v', 'g', 'H')
    def __init__(self, v, d=0, h=0, g=None, H=None):
        self.v = mp.mpf(v)
        self.g = {i:mp.mpf(x) for i,x in (g or {}).items() if x}
        self.H = {ij:mp.mpf(x) for ij,x in (H or {}).items() if x}
        if d:
            self.g[14]=mp.mpf(d)
        if h:
            self.H[14,14]=mp.mpf(h)
    @property
    def d(self):
        return self.g.get(14,mp.mpf(0))
    @property
    def h(self):
        return self.H.get((14,14),mp.mpf(0))
    @staticmethod
    def variable(v,index):
        require(type(index) is int and 0<=index<15, 'Independent15 coordinate')
        return Jet(v,g={index:mp.mpf(1)})
    @staticmethod
    def scalar(v):
        return v if isinstance(v, Jet) else Jet(v)
    def __add__(self, other):
        o=self.scalar(other)
        g,H=dict(self.g),dict(self.H)
        for i,x in o.g.items():
            g[i]=g.get(i,0)+x
        for ij,x in o.H.items():
            H[ij]=H.get(ij,0)+x
        return Jet(self.v+o.v,g=g,H=H)
    __radd__=__add__
    def __neg__(self):
        return Jet(-self.v,g={i:-x for i,x in self.g.items()},H={ij:-x for ij,x in self.H.items()})
    def __sub__(self, other):
        return self+-self.scalar(other)
    def __rsub__(self, other):
        return self.scalar(other)+-self
    def __mul__(self, other):
        o=self.scalar(other)
        g={i:x*o.v for i,x in self.g.items()};H={ij:x*o.v for ij,x in self.H.items()}
        for i,x in o.g.items():
            g[i]=g.get(i,0)+x*self.v
        for ij,x in o.H.items():
            H[ij]=H.get(ij,0)+x*self.v
        for i,x in self.g.items():
            for j,y in o.g.items():
                ij=(min(i,j),max(i,j))
                H[ij]=H.get(ij,0)+x*y*(2 if i==j else 1)
        return Jet(self.v*o.v,g=g,H=H)
    __rmul__=__mul__
    def unary(self, f, first, second):
        a,b=first(self.v),second(self.v);H={ij:a*x for ij,x in self.H.items()}
        items=sorted(self.g.items())
        for k,(i,x) in enumerate(items):
            for j,y in items[k:]:
                H[i,j]=H.get((i,j),0)+b*x*y
        return Jet(f(self.v),g={i:a*x for i,x in self.g.items()},H=H)
    def reciprocal(self):
        require(self.v!=0, 'Nonzero independent jet denominator')
        return self.unary(lambda x:1/x, lambda x:-1/x**2, lambda x:2/x**3)
    def sqrt(self):
        require(self.v>0, 'Positive independent squared normal norm')
        return self.unary(mp.sqrt, lambda x:1/(2*mp.sqrt(x)), lambda x:-1/(4*x*mp.sqrt(x)))
    def sin(self):
        return self.unary(mp.sin, mp.cos, lambda x:-mp.sin(x))
    def cos(self):
        return self.unary(mp.cos, lambda x:-mp.sin(x), lambda x:-mp.cos(x))
    def __truediv__(self, other):
        return self*self.scalar(other).reciprocal()
    def __rtruediv__(self, other):
        return self.scalar(other)*self.reciprocal()

def dot(a,b):
    return sum((x*y for x,y in zip(a,b)), Jet(0))

def mv(A,x):
    return tuple(dot(row,x) for row in A)

def mm(A,B):
    return tuple(tuple(dot(row,col) for col in zip(*B)) for row in A)

def tr(A):
    return tuple(zip(*A))

def cof(A):
    out=[]
    for i in range(3):
        rows=[k for k in range(3) if k!=i];row=[]
        for j in range(3):
            cols=[k for k in range(3) if k!=j]
            row.append((-1)**(i+j)*(A[rows[0]][cols[0]]*A[rows[1]][cols[1]]
                -A[rows[0]][cols[1]]*A[rows[1]][cols[0]]))
        out.append(tuple(row))
    return tuple(out)

def determinant(A):
    return dot(A[0],cof(A)[0])

def inverse(A):
    C,D = cof(A),determinant(A)
    return tuple(tuple(C[j][i]/D for j in range(3)) for i in range(3))

def cross(a,b):
    return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])

class Twist:
    def __init__(self,R,s,omega):
        self.R,self.s,self.omega = R,s,omega
        self.k = mp.sqrt(sum(w*w for w in omega))
        require(self.k>0, 'Authentic nonzero common twist rate')
        a,b,c = omega
        self.W = ((0,-c,b),(c,0,-a),(-b,a,0))
        self.W2 = numeric(mm(self.W,self.W))
    def map_gradient(self,x,lam):
        rho = mp.sqrt(sum(v*v for v in x))
        tau = (self.R**3-(rho/self.s)**3)/(3*self.R**2)
        g = tuple(-rho*v/(self.s**3*self.R**2) for v in x)
        v = cross(self.omega,x)
        d = tuple(tuple(v[i]*g[j] for j in range(3)) for i in range(3))
        angle = lam*(tau*self.k)
        a,b = angle.sin()/self.k,(1-angle.cos())/(self.k*self.k)
        Q = tuple(tuple(int(i==j)+a*self.W[i][j]+b*self.W2[i][j] for j in range(3)) for i in range(3))
        D = mm(Q,tuple(tuple(int(i==j)+lam*d[i][j] for j in range(3)) for i in range(3)))
        return mv(Q,x),D,tau,d,Q
    def scalar_component(self,i,x,lam):
        # Independently differentiable scalar Rodrigues expression, retained
        # for Cartesian and lambda mpmath.diff controls of the jet algebra.
        rho=mp.sqrt(sum(v*v for v in x));tau=(self.R**3-(rho/self.s)**3)/(3*self.R**2)
        axis=tuple(w/self.k for w in self.omega);c,s=mp.cos(lam*tau*self.k),mp.sin(lam*tau*self.k)
        turn=cross(axis,x);along=sum(u*v for u,v in zip(axis,x))
        return c*x[i]+s*turn[i]+(1-c)*axis[i]*along

def fresh_owner_rotation(owner):

        H={(3*owner+i,3*owner+j):a[r][c] for (i,j),a in second.items()}) for c in range(3)) for r in range(3))

def side(x,A0,Aold,l0,twist,lam,*,internal=None,vold=None,owner=None):
    y,D,tau,d,Q = twist.map_gradient(x,lam)
    A = mm(D,A0)
    arm=l0 if internal is None else tuple(a+internal*v for a,v in zip(l0,vold))
    transported = mv(mm(A,inverse(Aold)),arm)
    Y = tuple(u+v for u,v in zip(y,transported))
    return dict(material_position=y,original_gradient=A,attachment_position=Y,
        map_gradient=D,depth=tau,d=d,rotation=Q,owner=owner,internal_arm=arm)

def normal(A,N):
    w = mv(cof(A),N)
    length = dot(w,w).sqrt()
    return tuple(v/length for v in w),w,length

def cohesive(d,kn,C,T,Gamma):
    K = 1/(1/kn+C);g0=T/kn;gf=2*Gamma/T;onset=g0+C*T
    require(gf>onset>0, 'Actual source no-snapback loading law')
    if d.v<onset:
        gap = d/(1+C*kn);traction=K*d;damage=Jet(0);fracture=Jet(0)
        phi=K*d*d/2;recoverable=kn*gap*gap/2+C*traction*traction/2
        branch='intact_or_compressed'
    elif d.v>gf:
        gap=d;traction=Jet(0);damage=Jet(1);fracture=Jet(Gamma)
        phi=Jet(Gamma);recoverable=Jet(0);branch='fully_failed_open'
    else:
        require(onset<d.v<gf, 'A branch corner has no unique classical second derivative')
        gap=g0+(d-onset)*(gf-g0)/(gf-onset)
        traction=T*(gf-d)/(gf-onset);damage=gf/(gf-g0)*(1-g0/gap)
        fracture=Gamma*(gap-g0)/(gf-g0)
        delta=d-onset;phi=T*onset/2+T*delta-T*delta*delta/(2*(gf-onset))
        recoverable=kn*(1-damage)*gap*gap/2+C*traction*traction/2
        branch='monotone_softening'
    return dict(branch=branch,phi=phi,recoverable=recoverable,fracture=fracture,
        contact_gap=gap,traction=traction,damage=damage,onset=onset,failure=gf)

def exact_source_law(atom):
    source=json.loads(atom['complete_original_atom_history_json'])['original_material']
    law,columns=source['law'],source['columns']
    kn=rat(law['normal_stiffness_pa_m']);T=rat(law['tensile_strength_pa']);Gamma=rat(law['fracture_energy_j_m2'])
    C=rat(columns['sub_length_m'])/rat(columns['sub_normal_modulus_pa'])+rat(columns['over_length_m'])/rat(columns['over_normal_modulus_pa'])
    K=1/(1/kn+C)
    require(K==rat(atom['series_stiffness_exact_pa_m'][0]), 'Actual complete source series stiffness')
    return dict(kn=num(kn),C=num(C),T=num(T),Gamma=num(Gamma)),K

def extract_token(path,token):
    # Bytes are only observed from the old authority. The old mathematical
    # controls are neither imported nor called to restore this saved value.
    decoder=json.JSONDecoder()
    with Path(path).open('rb') as stream,mmap.mmap(stream.fileno(),0,access=mmap.ACCESS_READ) as mapped:
        pos=mapped.find(token.encode());require(pos>=0, 'Saved authority token '+token)
        pos+=len(token);extent=1<<20
        while extent<=1<<28:
            text=mapped[pos:min(len(mapped),pos+extent)].decode('utf-8',errors='ignore').lstrip()
            try:
                return decoder.raw_decode(text)[0]
            except json.JSONDecodeError:
                extent*=2
    raise AssertionError('Explicit256MiB saved token extraction bound')

    with Path(path).open('rb') as stream,mmap.mmap(stream.fileno(),0,access=mmap.ACCESS_READ) as mapped:
        pos=mapped.find(token);require(pos>=0, 'Saved complete authority array '+array_key)
        pos+=len(token)
        while pos<len(mapped):
            while mapped[pos] in b' \n\r\t,':
                pos+=1
            require(mapped[pos]!=ord(']'), 'Named saved authority row '+label)
            extent=1<<20
            while extent<=1<<28:
                text=mapped[pos:min(len(mapped),pos+extent)].decode('utf-8',errors='ignore')

                    row,end=decoder.raw_decode(text)

                except json.JSONDecodeError:
                    extent*=2
            else:
                raise AssertionError('Explicit256MiB saved array-row extraction bound')
            if row['label']==label:

            pos+=len(text[:end].encode('utf-8'))

def qbounds(raw):
    if type(raw) is dict and set(raw)=={'lo','hi'}:
        return rat(raw['lo']),rat(raw['hi'])
    exact=rat(raw)
    return exact,exact

def rational_interval(raw):
    lo,hi=qbounds(raw)
    require(lo<=hi,'Ordered exact rational certificate interval')
    return lo,hi

def rational_add(a,b):
    return a[0]+b[0],a[1]+b[1]

def rational_multiply(a,b):
    products=tuple(x*y for x in a for y in b)
    return min(products),max(products)

def rational_round(a,precision):
    """Enclose a rational interval on the declared relative binary grid.

    The inherited source-carrier compression adds sixteen decimal guard
    digits. 3322/1000 bounds log2(10) above and eight more binary guard
    bits define the pose grid. This verifier imports no native arithmetic.
    """
    maximum=max(abs(a[0]),abs(a[1]))
    if not maximum:
        return F(0),F(0)
    exponent=maximum.numerator.bit_length()-maximum.denominator.bit_length()
    below=(maximum.numerator < maximum.denominator*(1<<exponent)) if exponent>=0 else (
        maximum.numerator*(1<<(-exponent)) < maximum.denominator)
    exponent-=int(below)
    bits=(3322*(precision+16)+999)//1000+8
    shift=bits-exponent
    scale=F(1<<shift) if shift>=0 else F(1,1<<(-shift))
    lo,hi=a[0]*scale,a[1]*scale
    return F(lo.numerator//lo.denominator)/scale,F(-((-hi.numerator)//hi.denominator))/scale

def rational_matrix_product(a,b,precision):
    result=[]
    for i in range(3):
        row=[]
        for j in range(3):
            total=(F(0),F(0))
            for k in range(3):
                total=rational_add(total,rational_multiply(a[i][k],b[k][j]))
            row.append(rational_round(total,precision))
        result.append(tuple(row))
    return tuple(result)

    """Verify the short new certificate, not an old physics or audit main."""
    c=json.loads(canonical_certificate)
    p,work=c['output_precision_decimal_digits'],c['work_precision_decimal_digits']
    s,n=c['scaling_squaring_count'],c['taylor_term_count']

    divisor=F(1<<s)

    one=tuple(tuple((F(int(i==j)),)*2 for j in range(3)) for i in range(3))
    term,total=one,one

    for _ in range(s):

    return tuple(tuple(rational_round(v,p) for v in row) for row in total)

def polynomial_jet(coefficients,lam):
    result=Jet(value(coefficients[-1]))
    for coefficient in reversed(coefficients[:-1]):
        result=value(coefficient)+lam*result
    return result

def arbitrary_owner_rotation(theta):
    a,b,c=theta
    return mp.expm(mp.matrix(((0,-c,b),(c,0,-a),(-b,a,0))))

def numeric_contact_coordinate(ctx,atom,coords,*,energy=False):
    aid=atom['point_id'];lam=Jet(coords[14]);internal=list(coords[6:14])
    prepared=fixture_contact(ctx,atom,lam,internal)
    Ys,Yo=prepared['sub_side']['attachment_position'],prepared['over_side']['attachment_position']
    body_normal,_,_=normal(prepared['over_side']['original_gradient'],prepared['material_normal'])
    owners=[ctx['physical'][aid,side_name]['owner'] for side_name in ('sub','over')]
    matrices=[arbitrary_owner_rotation(coords[3*owner:3*owner+3]) for owner in owners]
    vectors=[matrix_*mp.matrix(numeric(Y)) for matrix_,Y in zip(matrices,(Ys,Yo))]
    n=matrices[1]*mp.matrix(numeric(body_normal))
    gap=sum(n[i]*(vectors[1][i]-vectors[0][i]) for i in range(3))-prepared['baseline']
    if not energy:
        return gap
    loading=cohesive(Jet(prepared['free_gap'].v-prepared['delta'].v+gap),**prepared['law'])
    return prepared['offset']+num(atom['original_area_m2'])*loading['phi'].v

def independent_derivative_controls(ctx,rows,counts):

                        [v+increment*int(k==i) for k,v in enumerate(coords)],energy=energy)
