def primitives():
    """Authenticate old stdlib definitions without calling any run/main."""
    helper = PARENT / 'baseline_899_preservation_source.txt'
    if hashlib.sha256(helper.read_bytes()).hexdigest() != PRESERVATION899_RAW:
        raise ValueError('Immutable899 stdlib primitive source changed')
    module = types.ModuleType('_private_907_preservation_primitives')
    module.__file__ = str(helper)
    exec(compile(helper.read_bytes(), str(helper), 'exec'), module.__dict__)
    prior, byte_checks = module.primitives()
    module._prior_definitions = prior
    return module, byte_checks


def project_modules():
    return {name: module for name, module in sys.modules.items()
            if name in ('tectonics', 'tests', 'analysis', 'scripts') or
            name.startswith(('tectonics.', 'tests.', 'analysis.', 'scripts.'))}

    b.exact(set(additional_source_sha256), set(ADDED_NAMES), 'Exactly four authorized907 source paths')
    requested = Path(output_path)
    b.require(requested.is_absolute() and requested.resolve() == requested and not requested.exists()
              and not requested.is_symlink() and requested.parent.is_dir()
              and requested.parent.resolve() == requested.parent and requested.suffix == '.json'
              and not requested.is_relative_to(ROOT) and
              (requested.is_relative_to(Path('/workspace')) or requested.is_relative_to(Path('/tmp'))),
              'Fresh canonical exclusive JSON output outside repository')

    producer_raw = b.digest(producer)

    declared = dict(frozen_prior)
    for name, expected in additional_source_sha256.items():
