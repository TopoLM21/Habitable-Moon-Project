def enc(value):
 return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def vd(value):return hashlib.sha256(enc(value)).hexdigest()
def read(path):return json.loads(Path(path).read_bytes())
def digest(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as stream:
  for block in iter(lambda:stream.read(1<<20),b''):h.update(block)
 return h.hexdigest()
def sealed(path,key):
 raw=read(path);body=dict(raw);assert body.pop(key)==vd(body),str(path);return raw
def exact(actual,expected,label):assert actual==expected,label
def literal(value):
 assert type(value) in (float,int),'literal source number'
 return F.from_float(value) if type(value) is float else F(value)
def frac(value):
 assert type(value) is list and len(value)==2 and all(type(x) is int for x in value)
 q=F(*value);assert [q.numerator,q.denominator]==value,'canonical rational';return q
def payload(value):
 if type(value) is F:return [value.numerator,value.denominator]
 if hasattr(value,'lo') and hasattr(value,'hi'):return {'lo':payload(value.lo),'hi':payload(value.hi)}
 if type(value) in (list,tuple):return [payload(x) for x in value]
 if type(value) is dict:return {k:payload(v) for k,v in value.items()}
 return value

def cross(first,second):
 return (first[1]*second[2]-first[2]*second[1],
         first[2]*second[0]-first[0]*second[2],

def mechanical_oracle(context,atom,basis,branch,increments):
 """Independent exact source-driver two-jet, without production imports.

 Receiver radial/rotation cancellation is performed symbolically.  Matrix
 bounds come from an independent high precision ordered quaternion history.
 """
 I=context['I'];order=context['order'];dim=len(order);assert dim==14
 assert basis in ('archival_old_reference','current_objective')
 assert branch in ('relaxed_zero_strength_shear','frozen_elastic_predictor')
 assert len(increments)==14 and all(type(x) is F for x in increments) and all(x==0 for x in increments[:6])
 tid=atom['trace_id'];point=atom['original_point'];material=atom['original_material'];state=atom['unaccepted_candidate_state']
 sub,over=material['subducting_plate'],material['overriding_plate']
 q=tuple(literal(x) for x in point['position']);frames=tuple(tuple(literal(x) for x in point[k]) for k in ('normal','down_dip'))
 Q=context['poses'][basis]
 mapped=lambda owner,v:tuple(dot(row,v) for row in Q[owner])
 qs,qo=mapped(sub,q),mapped(over,q)
 e=tuple(mapped(over,v) for v in frames)
 r=context['radius'];radial=context['radial'][basis][tid]
 wi=tuple(order.index(('radial',tid,side)) for side in ('sub','over'))
 rhos=(r+radial[0]+increments[wi[0]],r+radial[1]+increments[wi[1]])
 ps,po=scale(qs,rhos[0]),scale(qo,rhos[1]);D=minus(po,ps)
 h=tuple(rhos[1]*dot(frame,q)-dot(axis,ps) for frame,axis in zip(frames,e))
 # A radial-only query makes eta exactly linear, retaining the old source
 # spring anchor without subtracting rounded geometry values.
 eta=tuple(-dot(axis,qs)*increments[wi[0]]+dot(frame,q)*increments[wi[1]] for frame,axis in zip(frames,e))
 B=[];H=[]
 for frame,axis in zip(frames,e):
  b=[I(0) for _ in order];hh=[[I(0) for _ in order] for _ in order]
  lever=cross(ps,axis)
  for j in range(3):
   b[order.index(('owner',sub,('x','y','z')[j]))]=-lever[j]
   b[order.index(('owner',over,('x','y','z')[j]))]=lever[j]
  b[wi[0]]=-dot(axis,qs);b[wi[1]]=I(dot(frame,q))
  for i,a in enumerate(AXES):
   si=order.index(('owner',sub,('x','y','z')[i]));oi=order.index(('owner',over,('x','y','z')[i]))
   hh[si][wi[0]]=hh[wi[0]][si]=-dot(axis,cross(a,qs))
   hh[oi][wi[0]]=hh[wi[0]][oi]=-dot(cross(a,axis),qs)
   for j,c in enumerate(AXES):
    sj=order.index(('owner',sub,('x','y','z')[j]));oj=order.index(('owner',over,('x','y','z')[j]))
    hh[si][sj]=-dot(axis,sym(a,c,ps))
    hh[oi][oj]=-dot(sym(a,c,axis),ps)
    hh[si][oj]=hh[oj][si]=-dot(cross(c,axis),cross(a,ps))
  B.append(tuple(b));H.append(tuple(tuple(row) for row in hh))
 kn=literal(material['law']['normal_stiffness_pa_m']);kt=literal(material['law']['tangential_stiffness_pa_m'])
 col=material['columns']
 series=(1/(1/kn+literal(col['sub_length_m'])/literal(col['sub_normal_modulus_pa'])+literal(col['over_length_m'])/literal(col['over_normal_modulus_pa'])),
         1/(1/kt+literal(col['sub_length_m'])/literal(col['sub_shear_modulus_pa'])+literal(col['over_length_m'])/literal(col['over_shear_modulus_pa'])))
 effective=(series[0],F(0) if branch=='relaxed_zero_strength_shear' else series[1])
 t0=tuple(literal(state[k]) for k in ('traction_normal_pa','traction_shear_pa'))
 assert t0[0]<0 and t0[1]==0
 traction=tuple(I(t)+k*x for t,k,x in zip(t0,effective,eta));assert traction[0].hi<0,'declared query remains compressed'
 free0=tuple(literal(state[k]) for k in ('gap_free_m','jump_free_m'))
 free=tuple(I(x)+dx for x,dx in zip(free0,eta))
 area=literal(atom['original_area_m2']);components=tuple(literal(state[k]) for k in ('contact_energy_j','bulk_energy_j'));energy0=sum(components,F(0))
 energy=I(energy0)+area*sum((I(t)*x+k*x*x/2 for t,k,x in zip(t0,effective,eta)),I(0))
 gradient=tuple(area*sum((t*b[i] for t,b in zip(traction,B)),I(0)) for i in range(dim))
 hessian=tuple(tuple(area*sum((k*b[i]*b[j]+t*hh[i][j] for k,b,t,hh in zip(effective,B,traction,H)),I(0)) for j in range(dim)) for i in range(dim))
 force=scale(add(scale(e[0],traction[0]),scale(e[1],traction[1])),area)
 ts=cross(ps,force);force_to=cross(po,scale(force,F(-1)));couple=cross(D,force)
 reactions={'force_on_sub_n':force,'force_on_over_n':scale(force,F(-1)),
  'torque_on_sub_nm':ts,'torque_on_over_nm':scale(ts,F(-1)),
  'force_torque_on_over_nm':force_to,'carrier_couple_on_over_nm':couple,
  'radial_force_n':(dot(qs,force),-dot(qo,force))}
 rounding_t=(t0[0]-series[0]*free0[0],t0[1]-series[1]*(free0[1]-literal(state['plastic_slip_m'])))
 rounding_e=energy0-area*sum((t*t/(2*k) for t,k in zip(t0,series)),F(0))
 binary=[]
 for component,contact_key in (('normal','normal_stiffness_pa_m'),('shear','tangential_stiffness_pa_m')):
  kc=material['law'][contact_key]
  columnk=1./sum(col[side+'_length_m']/col[side+'_'+component+'_modulus_pa'] for side in ('sub','over'))
  binary.append(literal(kc/(1.+kc/columnk)))
 return dict(q=q,frames=frames,sub=sub,over=over,area=area,ps=ps,po=po,qs=qs,qo=qo,e=e,h=h,
  eta=eta,B=tuple(B),H=tuple(H),free=free,free0=free0,t0=t0,traction=traction,series=series,effective=effective,
  components=components,energy0=energy0,energy=energy,gradient=gradient,hessian=hessian,reactions=reactions,
  rounding_t=rounding_t,rounding_e=rounding_e,binary=tuple(binary),plastic=eta[1] if branch=='relaxed_zero_strength_shear' else I(0))


def parse_interval(raw,context):
 assert type(raw) is dict and set(raw)=={'lo','hi'}
 return context['I'](frac(raw['lo']),frac(raw['hi']))
def contains(raw,expected,context,label):
 actual=parse_interval(raw,context)
 if not hasattr(expected,'lo'):expected=context['I'](expected)
 assert actual.lo<=expected.lo<=expected.hi<=actual.hi,label
 return actual
def contains_tree(raw,expected,context,label):
 if isinstance(expected,(tuple,list)):
  assert type(raw) is list and len(raw)==len(expected),label+' shape'
  return tuple(contains_tree(x,y,context,label+':'+str(i)) for i,(x,y) in enumerate(zip(raw,expected)))
 return contains(raw,expected,context,label)
