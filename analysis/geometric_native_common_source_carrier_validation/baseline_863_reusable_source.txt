therefore receives the explicitly disclosed frozen863 projection. The complete
current source map, including declared additions, is checked independently
before and after that reader and every old source hash is checked directly.

def enc(x):
    return json.dumps(x, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()


def vd(x):
    return hashlib.sha256(enc(x)).hexdigest()


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def sealed(path, key):
    raw = read(path)
    body = dict(raw)
    assert body.pop(key) == vd(body), str(path)
    return raw


def load(path, name, pin):
    assert digest(path) == pin, str(path)
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module
    exec(compile(Path(path).read_bytes(), str(path), 'exec'), module.__dict__)
    return module


def source_map():
    return {str(p.relative_to(ROOT)): digest(p)
            for folder in ('tectonics', 'tests', 'scripts', 'analysis')
            for p in sorted((ROOT / folder).rglob('*.py'))}


def normalized_declared_paths(paths):
    assert type(EXPECTED_ADDED_COUNT) is int and EXPECTED_ADDED_COUNT > 0, 'Freeze exact addition count first'
    assert not isinstance(paths, (str, bytes)), 'Pass an iterable of explicit path strings'
    names = tuple(paths)
    assert len(names) == len(set(names)) == EXPECTED_ADDED_COUNT, 'Wrong declared addition count'
    for name in names:
        assert isinstance(name, str) and name, 'Nonempty path strings required'
        path = PurePosixPath(name)
        assert not path.is_absolute() and str(path) == name, name
        assert len(path.parts) >= 2 and '..' not in path.parts, name
        assert path.parts[0] in ('tectonics', 'tests', 'scripts', 'analysis'), name
        assert path.suffix == '.py', name
    assert frozenset(names) == EXPECTED_ADDED_PATHS, 'Use the exact four authorized carrier source paths'
    return frozenset(names)
