def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
                      ensure_ascii=False, allow_nan=False).encode()


def value_digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()


def require(condition, label):
    if not condition:
        raise ValueError(label)


def exact(actual, expected, label):
    require(type(actual) is type(expected) and actual == expected, label)
    if type(actual) in (dict, list, tuple):
        require(encoded(actual) == encoded(expected), label + ' complete literal encoding')


def regular(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path and not path.is_symlink()
            and path.is_file(), 'Canonical regular file ' + str(path))
    return path


def digest(path):
    h = hashlib.sha256()
    with regular(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def pin(path):
    path = regular(path)
    return dict(bytes=path.stat().st_size, sha256=digest(path))


def sealed(path, key):
    value = json.loads(regular(path).read_bytes())
    require(type(value) is dict and key in value, 'Complete sealed object ' + str(path))
    exact(value[key], value_digest({k: v for k, v in value.items() if k != key}),
          'Complete selfseal ' + str(path))
    return value


def fresh(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path and not path.exists()
            and not path.is_symlink() and path.parent.is_dir()
            and path.parent.resolve() == path.parent, 'Fresh canonical exclusive output ' + str(path))
    return path


def write_new(path, value):
    with fresh(path).open('xb') as stream:
        stream.write(encoded(value) + b'\n')


def copy_new(source, target):
    source = regular(source)
    expected = pin(source)
    with fresh(target).open('xb') as stream, source.open('rb') as incoming:
        for block in iter(lambda: incoming.read(1 << 20), b''):
            stream.write(block)
    exact(pin(target), expected, 'Exact exclusive archived copy ' + str(target))
    return expected


def all_files(directory):
    return {str(path.relative_to(directory)) for path in directory.rglob('*') if path.is_file()}


def meaningful_files(directory):
    return {str(path.relative_to(directory)) for path in directory.rglob('*')
            if path.is_file() and '__pycache__' not in path.parts and path.suffix != '.pyc'}


def source_map():
    return {str(path.relative_to(ROOT)): digest(path)
            for directory in ('tectonics', 'tests', 'scripts', 'analysis')
            for path in sorted((ROOT / directory).rglob('*.py'))}


def project_modules():
    return {name: module for name, module in sys.modules.items()
            if name in ('tectonics', 'tests', 'analysis', 'scripts') or
            name.startswith(('tectonics.', 'tests.', 'analysis.', 'scripts.'))}


def unchanged_modules(before):
    after = project_modules()
    exact(set(after), set(before), 'Caller project module names preserved')
    require(all(after[name] is module for name, module in before.items()),
            'Caller project module identities preserved')


def document_pins():

def template_literals():
    exact(digest(TEMPLATE), TEMPLATE_RAW, 'Reviewed reader template raw')
    tree = ast.parse(TEMPLATE.read_bytes())
    imports = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imports.extend(alias.name for alias in node.names)
        elif isinstance(node, ast.ImportFrom):
            imports.append(node.module)
    require(set(imports) <= {'pathlib', 'argparse', 'hashlib', 'json', 'sys', 'time', 'types'},
            'Reviewed reader has stdlib imports only')
    values = {}
    for name in ('EXPECTED_NATIVE', 'REMAINING_CONDITIONS'):
        matches = [node.value for node in tree.body if isinstance(node, ast.Assign)
                   and any(isinstance(target, ast.Name) and target.id == name for target in node.targets)]
        exact(len(matches), 1, 'Exactly one reviewed reader literal ' + name)
        values[name] = ast.literal_eval(matches[0])
    return values

    require(release.is_dir() and release.resolve() == release and not release.is_symlink(),
            'Canonical completed release directory')
    return release

    exact(set(gate['output_raw_sha256']), set(PAYLOADS), 'Complete four output raw map')

def manifest_files():
    files = {str(path.relative_to(ROOT)): pin(path) for path in sorted(PARENT.rglob('*'))
             if path.is_file() and '__pycache__' not in path.parts and path.suffix != '.pyc'
             and path != PARENT / 'artifact_manifest.json'}
    for name in ADDED:
        files[name] = pin(ROOT / name)
    exact({name for name in files if name.endswith('.py')}, set(ADDED), 'Exactly four component Python rows')
    document_names = {str(ROOT / name) for name in DOCUMENT_NAMES}
    require(not any(str(ROOT / name) in document_names for name in files),
            'Mutable handoff documents excluded from immutable dependency rows')
    return files


def check_manifest(manifest):
    exact(manifest_files(), manifest['files'], 'Complete current899 component raw map unchanged')
    expected_parent = {str((ROOT / name).relative_to(PARENT)) for name in manifest['files']
                       if (ROOT / name).is_relative_to(PARENT)}
    exact(meaningful_files(PARENT), expected_parent | {'artifact_manifest.json'}, 'Strict complete899 P namespace')
    exact(len(manifest['files']), manifest['closed_component_file_count'], 'Complete899 manifest row count')


def reader_source(manifest):
    template_literals()
    text = TEMPLATE.read_text()
    raw_marker = '__ROOT_FILL_CURRENT899_MANIFEST_RAW_SHA256__'
    seal_marker = '__ROOT_FILL_CURRENT899_MANIFEST_SELFSEAL__'
    count_marker = 'MANIFEST_FILE_COUNT = None  # ROOT_FILL: exact closed current899 manifest file count.'
    for marker in (raw_marker, seal_marker, count_marker):
        exact(text.count(marker), 1, 'Exactly one unresolved reviewed reader marker')
    text = text.replace(raw_marker, digest(PARENT / 'artifact_manifest.json'))
    text = text.replace(seal_marker, manifest['manifest_sha256'])
    text = text.replace(count_marker, 'MANIFEST_FILE_COUNT = ' + str(manifest['closed_component_file_count']))
    require('__ROOT_FILL_' not in text, 'All three reader placeholders filled')
    compile(text, str(FINAL / 'readback_source.txt'), 'exec')
    return text.encode()


def prepare_manifest(args):
    started = time.perf_counter()
    modules_before = project_modules()
    require(args.release is not None and args.gate_console is not None and args.gate_stderr is not None,
            'prepare_manifest requires --release, --gate-console, --gate-stderr')
    exact(args.gate_exit_code, 0, 'Actual gate process exit0 required')
    exact(args.audit_exit_code, 0, 'Actual audit process exit0 required')
    require(not (PARENT / 'artifact_manifest.json').exists() and not FINAL.exists()
            and not FINAL.is_symlink(), 'Fresh899 manifest and final directory only')
    gate_console = regular(Path(args.gate_console))
    gate_stderr = regular(Path(args.gate_stderr))
    require(not gate_console.is_relative_to(ROOT) and not gate_stderr.is_relative_to(ROOT),
            'True outer gate captures outside repository')
    producer = regular(Path(__file__).absolute())
    producer_raw = digest(producer)
    docs = document_pins()
    gate, audit, frozen = completed_component(args.release, gate_console, gate_stderr)
    copies = [(STAGE / name, PARENT / name) for name in BASELINE_NAMES]
    copies.extend(((TEMPLATE, PARENT / 'final899_readback_template.txt'),
                   (producer, PARENT / 'close_evidence_source.txt'),
                   (gate_console, PARENT / 'gate_actual_console.log'),
                   (gate_stderr, PARENT / 'gate_actual_stderr.log')))
    input_pins = {str(source): pin(source) for source, target in copies}
    before_path = STAGE / 'closure_prepare_prior895_before.json'
    after_path = STAGE / 'closure_prepare_prior895_after.json'
    fact_path = STAGE / 'prepare_manifest_actual_facts.json'
    for path in (before_path, after_path, fact_path, PARENT / 'artifact_manifest.json',
                 PARENT / 'closure_prior895_before.json', PARENT / 'closure_prior895_after.json'):
        fresh(path)
    for source, target in copies:
        fresh(target)
    before = preservation(gate['additional_source_sha256'], before_path)
    for source, target in copies:
        copy_new(source, target)
    copy_new(before_path, PARENT / 'closure_prior895_before.json')
    after = preservation(gate['additional_source_sha256'], after_path)
    copy_new(after_path, PARENT / 'closure_prior895_after.json')
    for name, expected in input_pins.items():
        exact(pin(Path(name)), expected, 'Closure input unchanged AFTER archival copy')
    exact(document_pins(), docs, 'Final documents unchanged during component closure')
    exact(source_map(), frozen, 'Full actual899 sources unchanged during component closure')
    exact(digest(producer), producer_raw, 'Executing closure source unchanged')
    completed_component(args.release, PARENT / 'gate_actual_console.log', PARENT / 'gate_actual_stderr.log')
    files = manifest_files()
    manifest = dict(
        format='native-common-cold-continuum-energy-component-manifest-1',
        status='PASS_COMPONENT_ARTIFACTS_CLOSED', source_count=899, physical_scope=SCOPE,
        current_release=args.release, closed_component_file_count=len(files), files=files,
        gate_report_selfseal=gate['report_sha256'], independent_audit_selfseal=audit['audit_sha256'],
        document_raw_sha256=docs,
        before_closure_preservation_selfseal=before['preservation_sha256'],
        after_closure_preservation_selfseal=after['preservation_sha256'],
        executing_closure_source_raw_sha256=producer_raw,
    )
    manifest['manifest_sha256'] = value_digest(manifest)
    write_new(PARENT / 'artifact_manifest.json', manifest)
    check_manifest(manifest)
    ready_reader = reader_source(manifest)
    FINAL.mkdir(exist_ok=False)
    with fresh(FINAL / 'readback_source.txt').open('xb') as stream:
        stream.write(ready_reader)
    exact(all_files(FINAL), {'readback_source.txt'}, 'Ready final directory contains reader source only')
    exact(document_pins(), docs, 'Final documents unchanged AFTER manifest closure')
    exact(source_map(), frozen, 'Full actual899 sources unchanged AFTER manifest closure')
    unchanged_modules(modules_before)
    facts = dict(
        status='PASS_899_MANIFEST_CLOSED_READER_READY', source_count=899,
        current_release=args.release, manifest_raw_sha256=digest(PARENT / 'artifact_manifest.json'),
        manifest_selfseal=manifest['manifest_sha256'], manifest_file_count=len(files),
        reader_path=str(FINAL / 'readback_source.txt'), reader_raw_sha256=digest(FINAL / 'readback_source.txt'),
        documents_sha256=docs, gate_exit_code=args.gate_exit_code, audit_exit_code=args.audit_exit_code,
        baseline_archived_file_count=len(BASELINE_NAMES), project_imports=False,
        math_or_tests_or_solver_or_quadrature_reexecuted=False,
        executing_closure_source_raw_sha256=producer_raw, seconds=time.perf_counter() - started,
    )
    write_new(fact_path, facts)
    print(encoded(facts).decode())
    return facts
