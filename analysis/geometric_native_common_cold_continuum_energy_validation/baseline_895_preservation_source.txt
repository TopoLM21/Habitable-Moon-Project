def primitives():
    """Authenticate old stdlib definitions without calling any run/main."""
    helper = PARENT / 'baseline_891_preservation_source.txt'
    if hashlib.sha256(helper.read_bytes()).hexdigest() != PRESERVATION891_RAW:
        raise ValueError('Immutable891 stdlib primitive source changed')
    module = types.ModuleType('_private_899_preservation_primitives')
    module.__file__ = str(helper)
    exec(compile(helper.read_bytes(), str(helper), 'exec'), module.__dict__)
    prior, byte_checks = module.primitives()
    module._prior_definitions = prior
    return module, byte_checks


def project_modules():
    return {name: module for name, module in sys.modules.items()
            if name in ('tectonics', 'tests', 'analysis') or
            name.startswith(('tectonics.', 'tests.', 'analysis.'))}

def run(additional_source_sha256, *, output_path):
    """Write a fresh external receipt for immutable895 plus authorized four899."""
    started = time.perf_counter()
    modules_before = project_modules()
    old, b = primitives()
    b.require(type(additional_source_sha256) is dict, 'Literal additional four source map')
    b.exact(set(additional_source_sha256), set(ADDED_NAMES), 'Exactly four authorized899 source paths')
    requested = Path(output_path)
    b.require(requested.is_absolute() and requested.resolve() == requested and not requested.exists()
              and not requested.is_symlink() and requested.parent.is_dir()
              and requested.parent.resolve() == requested.parent and requested.suffix == '.json'
              and not requested.is_relative_to(ROOT) and
              (requested.is_relative_to(Path('/workspace')) or requested.is_relative_to(Path('/tmp'))),
              'Fresh canonical exclusive JSON output outside repository')
    producer = b.regular(Path(__file__).resolve())
    producer_raw = b.digest(producer)
    frozen895, baseline, pins = preserved_inputs(old, b)
    declared = dict(frozen895)
    for name, expected in additional_source_sha256.items():
        b.require(name not in frozen895, 'Authorized899 source genuinely new')
        b.pin(pins, ROOT / name, expected)
        declared[name] = expected
    b.exact(b.source_map(), declared, 'Actual899 equals immutable895 plus exactly declared four')
    b.pin(pins, producer, producer_raw)
    b.require(str(requested) not in pins, 'Fresh output disjoint from preserved dependencies')
    after_frozen, after_baseline, after_pins = preserved_inputs(old, b)
    b.exact(after_frozen, frozen895, 'Saved895 authority unchanged AFTER')
    b.exact(after_baseline, baseline, 'Fresh pre-new-source895 baseline unchanged AFTER')
    for name, expected in pins.items():
        if name in after_pins:
            b.exact(after_pins[name], expected, 'Preserved dependency unchanged AFTER ' + name)
        else:
            b.exact(b.digest(name), expected, 'Added source/producer unchanged AFTER ' + name)
    b.exact(b.source_map(), declared, 'Complete actual899 source map unchanged AFTER')
    modules_after = project_modules()
    b.exact(set(modules_after), set(modules_before), 'Caller project module names preserved')
    b.require(all(modules_after[name] is module for name, module in modules_before.items()),
              'Caller project module identities preserved')
    result = dict(
        format='native-common-cold-continuum-energy-prior895-preservation-1',
        status='PASS_IMMUTABLE895_WITH_EXACT_FOUR_DECLARED_SOURCE_ADDITIONS',
        prior_python_source_count=895, current_python_source_count=899,
        prior_python_source_map_canonical_sha256=b.value_digest(frozen895),
        current_python_source_map_canonical_sha256=b.value_digest(declared),
        additional_python_source_count=4, additional_python_source_sha256=additional_source_sha256,
        all895_python_sources_unchanged=True, complete_actual_source_map_checked_before_after=True,
        prior_895_metadata_is_authenticated_saved_readback_only=True,
        fresh_pre_new_source_895_readback_path=str(STAGE / 'fresh895_readback.json'),
        fresh_pre_new_source_895_readback_raw_sha256=FRESH_REPORT_RAW,
        fresh_pre_new_source_895_readback_selfseal=FRESH_REPORT_SEAL,
        fresh_pre_new_source_895_readback_seconds=baseline['seconds'],
        preserved_prior883_raw_dependency_pin_count=341,
        recorded_raw_dependency_pin_count=len(pins), dependency_raw_sha256=pins,
        preserved_current895_manifest_file_count=50,
        preserved_current895_manifest_raw_sha256=MANIFEST_RAW,
        preserved_current895_manifest_selfseal=MANIFEST_SEAL,
        preserved_current895_final_readback_archive_file_count=4,
        preserved_current895_final_readback_archive_raw_sha256=FINAL_ARCHIVE_RAW,
        preserved_prior891_manifest_file_count=44, preserved_prior891_final_readback_archive_file_count=4,
        preserved_prior887_manifest_file_count=41, preserved_prior887_final_readback_archive_file_count=4,
        authoritative_ancestry_release_closure_count=11,
        preserved_current895_release_closed_file_count=5,
        temporary_dependency_copy_association_count=15,
        current_tmp_inputs_restored_by_this_guard=False,
        pre_baseline_authenticated_tmp_alias_restoration_count=0,
        project_modules_imported_reloaded_or_replaced=False,
        caller_existing_project_module_count=len(modules_before),
        prior_reader_or_auditor_or_preservation_main_reexecuted=False,
        math_or_solver_or_test_or_quadrature_reexecuted=False,
        executing_preservation_source_raw_sha256=producer_raw,
        seconds=time.perf_counter() - started,
    )
    result['preservation_sha256'] = b.value_digest(result)
    with requested.open('xb') as stream:
        stream.write(b.encoded(result) + b'\n')
    return result
