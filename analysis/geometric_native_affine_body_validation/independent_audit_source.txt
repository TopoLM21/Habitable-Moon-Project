def green_replay(R,E,G,q,Q):
 nu=E/(2*G)-1;lam=2*G*nu/(1-2*nu);assert -1<nu<F(1,2) and E>0 and G>0
 bases=tuple(ms(b,1/R) for b in BASIS)
 Fbody=eye()
 for x,b in zip(q,bases):Fbody=madd(Fbody,ms(b,x))
 deform=mm(Q,Fbody);assert mm(trans(Q),Q)==eye() and det(Q)==1
 green=ms(madd(mm(trans(deform),deform),ms(eye(),-1)),F(1,2))
 stress=madd(ms(green,2*G),ms(eye(),lam*tr(green)))
 energy=G*inner(green,green)+lam*tr(green)**2/2
 variations=tuple(mm(Q,b) for b in bases)
 dgreen=tuple(ms(madd(mm(trans(v),deform),mm(trans(deform),v)),F(1,2)) for v in variations)
 gradients=tuple(inner(stress,d) for d in dgreen)
 hessian=[]
 for i in range(6):
  row=[]
  for j in range(6):
   second=ms(madd(mm(trans(variations[i]),variations[j]),mm(trans(variations[j]),variations[i])),F(1,2))
   row.append(2*G*inner(dgreen[i],dgreen[j])+lam*tr(dgreen[i])*tr(dgreen[j])+inner(stress,second))
  hessian.append(tuple(row))
 hessian=tuple(hessian);assert hessian==trans(hessian)
 stiffness=tuple(tuple((2*G*inner(a,b)+lam*tr(a)*tr(b))/R**2 for b in BASIS) for a in BASIS)
 return dict(Fbody=Fbody,F=deform,green=green,S=stress,P=mm(deform,stress),energy=energy,J=det(deform),
  grad=gradients,hess=hessian,K=stiffness,nu=nu,lam=lam,variation=variations,dgreen=dgreen)


def check_bulk(row,V,R,E,G):

  result=green_replay(R,E,G,coords,matrix(r['spatial_rotation']));assert result['J']>0
  for name,value in (('body_deformation_gradient',result['Fbody']),('deformation_gradient',result['F']),
   ('green_strain',result['green']),('second_piola_stress_pa',result['S']),('first_piola_stress_pa',result['P'])):
   assert matrix(r[name])==value,('full tensor',name,row['component_id'])
  assert frac(r['energy_density_reference_j_m3'])==result['energy'] and frac(r['volume_jacobian'])==result['J']
  same_bounds(r['physical_volume_bounds_m3'],scale(V,result['J']));same_bounds(r['energy_bounds_j'],scale(V,result['energy']))
  assert len(r['generalized_gradient_bounds_n'])==6
  for x,value in zip(r['generalized_gradient_bounds_n'],result['grad']):same_bounds(x,scale(V,value))
  same_imatrix(r['generalized_hessian_bounds_n_m'],tuple(tuple(scale(V,x) for x in a) for a in result['hess']))
  if d is None:
   assert r['first_directional_derivative_bounds_j'] is None and r['second_directional_derivative_bounds_j'] is None
  else:
   first=sum((x*y for x,y in zip(result['grad'],d)),F(0));second=sum((d[i]*result['hess'][i][j]*d[j] for i in range(6) for j in range(6)),F(0))

   # A separate exact quartic energy polynomial in t checks directional work.

   e0=result['green'];e1=ms(madd(mm(trans(variation),result['F']),mm(trans(result['F']),variation)),F(1,2));e2=ms(mm(trans(variation),variation),F(1,2))
   coefficient1=2*G*inner(e0,e1)+result['lam']*tr(e0)*tr(e1)

   assert coefficient1==first and 2*coefficient2==second
 assert matrix(row['bulk_stiffness_per_reference_volume_n_m4'])==result['K']
 same_imatrix(row['bulk_stiffness_bounds_n_m'],tuple(tuple(scale(V,x) for x in a) for a in result['K']))

def dec(x):
 if type(x) is F:return D(x.numerator)/D(x.denominator)
 if type(x) is float:return D.from_float(x)
 return D(x)
def ddot(a,b):return sum((x*y for x,y in zip(a,b)),D(0))
def cross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])
def unit(v):
 n=ddot(v,v).sqrt();assert n>0;return tuple(x/n for x in v)
def sincos(x):
 s=st=x;c=ct=D(1)
 for n in range(1,600):
  st *= -x*x/D((2*n)*(2*n+1));ct *= -x*x/D((2*n-1)*(2*n));s+=st;c+=ct
  if max(abs(st),abs(ct))<D('1e-170'):return s,c
 raise AssertionError('Decimal trig did not converge')
def rotate(v,a,tau=D(1)):
 n=ddot(a,a).sqrt()
 if not n:return v
 axis=tuple(x/n for x in a);s,c=sincos(n*tau);av=cross(axis,v);p=ddot(axis,v)
 return tuple(v[i]*c+av[i]*s+axis[i]*p*(1-c) for i in range(3))
def rotations(v,rows):
 for row in rows:
  a=tuple(dec(F.from_float(row['dt_myr'])*F.from_float(w)) for w in row['omega_rad_per_myr']);v=rotate(v,a)
 return v

def plane(p):return rotations(tuple(dec(x) for x in p['normal']),p['rotations'])
def ray(p):
 if p['kind']=='cross':v=tuple(D(p['sign'])*x for x in cross(plane(p['planes'][0]),plane(p['planes'][1])))
 else:
  assert len(p['reference'])==3;v=tuple(dec(x) for x in p['reference'])
 return rotations(v,p['rotations'])
def source_frame(carrier):
 arc=carrier['exact_initial_shared_arc'];u=unit(ray(arc['start']));endpoint=unit(ray(arc['end']));pn=unit(plane(arc['plane']));t=unit(cross(pn,u))
 if ddot(t,endpoint)<0:t=tuple(-x for x in t)
 n=cross(u,t);receiving=unit(plane(carrier['receiver_inward_plane']));sign=1 if ddot(n,receiving)>0 else -1
 return u,t,n,sign

def gauss(n):
 nodes=[]
 for index in range(1,n//2+1):
  x=D.from_float(math.cos(math.pi*(index-.25)/(n+.5)))
  for _ in range(30):
   p0,p1=D(1),x
   for k in range(2,n+1):p0,p1=p1,((2*k-1)*x*p1-(k-1)*p0)/k
   deriv=n*(x*p1-p0)/(x*x-1);change=p1/deriv;x-=change
   if abs(change)<D('1e-160'):break
  else:raise AssertionError('Legendre node did not converge')
  weight=2/((1-x*x)*deriv*deriv);nodes.extend(((x,weight),(-x,weight)))
 return tuple(nodes)


def face_decimal_audit(raw,carrier,A,R,top,depth):
 with localcontext() as context:
  context.prec=180
  u,t,n,orientation=source_frame(carrier);angle=bounds(carrier['native_angle_bounds_radians'])
  theta=dec((angle[0]+angle[1])/2);theta_hi=dec(angle[1]);end_error=dec((angle[1]-angle[0])/2)
  sr,cr=sincos(theta/2);U=tuple(x*cr+y*sr for x,y in zip(u,t));T=tuple(-x*sr+y*cr for x,y in zip(u,t))
  rate=tuple(dec(x) for x in A);rate_norm=ddot(rate,rate).sqrt();projection=ddot(rate,T)
  radius=dec(R);top_cube=dec(top**3);L=dec(depth);results=[D(0)]*6;count=32
  for node,weight in gauss(count):
   z=L*(node+1)/2;cube=top_cube-3*radius**2*z;tau=(radius**3-cube)/(3*radius**2)
   SU=rotate(U,rate,tau);SN=rotate(n,rate,tau)
   for k,(i,j) in enumerate(((0,0),(1,1),(2,2),(0,1),(0,2),(1,2))):
    mixed=SN[i]*SU[i] if i==j else SN[i]*SU[j]+SN[j]*SU[i]
    quadratic=SU[i]**2 if i==j else 2*SU[i]*SU[j]
    results[k]+=weight*(radius*mixed-cube/radius*projection*quadratic)/2
  role=raw['gap_sign'];centres=tuple(D(role*orientation)*theta*L*x for x in results)
  frequency=2*rate_norm*L;norder=2*count
  derivative_bound=radius*frequency**norder+rate_norm/radius*(top_cube*frequency**norder+norder*3*radius**2*L*frequency**(norder-1))
  gauss_factor=D(math.factorial(count))**4/(D(2*count+1)*D(math.factorial(2*count))**3)
  depth_error=theta_hi*L*gauss_factor*derivative_bound
  magnitude=L*(radius+top_cube*rate_norm/radius)
  angular_error=theta_hi**2*L*(radius+3*top_cube*rate_norm/radius)/4+end_error*magnitude
  # Native128 frame enclosures and Decimal180 reconstruction errors are tiny
  # against the full-angle bound. This allowance is deliberately conservative.
  arithmetic_error=theta_hi*magnitude*D('1e-105')
  total_error=depth_error+angular_error+arithmetic_error
  for k,(v,x) in enumerate(zip(raw['generalized_normal_force_bounds_n_per_pa'],centres)):
   lo,hi=map(dec,bounds(v));assert max(lo,x-total_error)<=min(hi,x+total_error),(raw['family_id'],raw['bank'],k,'independent Rodrigues face integral')
  ss,cc=sincos(theta);delta=tuple(x*(cc-1)+y*ss for x,y in zip(u,t))
  analytic=-D(role*orientation)*ddot(rate,delta)*dec((top**3*depth-F(3,2)*R**2*depth**2)/R)
  actual=isum(bounds(x) for x in raw['generalized_normal_force_bounds_n_per_pa'][:3]);assert dec(actual[0])-total_error<=analytic<=dec(actual[1])+total_error
  strongest=max(range(3,6),key=lambda k:abs(centres[k]));assert abs(centres[strongest])>total_error
  return dict(family_id=raw['family_id'],bank=raw['bank'],source_receiving_orientation=orientation,
   strongest_shear_index=strongest,strongest_shear_integral_m2=float(centres[strongest]),
   independent_error_bound_m2=float(total_error),full_six_mode_rodrigues_quadrature_overlap=True,
   analytic_dilatation_trace_checked=True,commutator_coefficients_reused=False)
