from pathlib import Path
import argparse,hashlib,json,time,types
ROOT=Path('/workspace/Habitable-Moon-Project');P=ROOT/'analysis/geometric_native_common_hot_receiving_validation'
F=ROOT/'analysis/geometric_native_common_hot_receiving_final_readback';STAGE=Path('/workspace/hot_receiving_baseline_20261007')

DOCS=('CLOUD_HANDOFF.md','GEOMETRIC_NATIVE_COMMON_HOT_RECEIVING_IMPLEMENTATION.md')
def enc(x):return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def row(p):return dict(bytes=Path(p).stat().st_size,sha256=digest(p))
def write(p,x):
 with Path(p).open('xb') as f:f.write(enc(x)+b'\n')
def sealed(p,k):
 x=json.loads(Path(p).read_bytes());assert hashlib.sha256(enc({a:b for a,b in x.items() if a!=k})).hexdigest()==x[k];return x
def files(p):return {str(x.relative_to(p)) for x in p.rglob('*') if x.is_file() and '__pycache__' not in x.parts and x.suffix!='.pyc'}
def manifest_rows():
 result={str(x.relative_to(ROOT)):row(x) for x in P.rglob('*') if x.is_file() and '__pycache__' not in x.parts and x.suffix!='.pyc' and x.name!='artifact_manifest.json'}
 result.update({name:row(ROOT/name) for name in ADDED});return result

 m=types.ModuleType('_909_close_guard');m.__file__=str(p);exec(compile(p.read_bytes(),str(p),'exec'),m.__dict__);return m

 for source,name in ((Path(__file__).resolve(),'close_evidence_source.txt'),):
  with (P/name).open('xb') as f:f.write(source.read_bytes())
 docs={str(ROOT/name):digest(ROOT/name) for name in DOCS}
 g=guard();before=g.run(gate['additional_source_sha256'],STAGE/'closure_before.json')
