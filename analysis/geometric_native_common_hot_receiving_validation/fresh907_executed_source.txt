def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
                      ensure_ascii=False, allow_nan=False).encode()


def require(condition, label):
    if not condition:
        raise ValueError(label)


def exact(actual, expected, label):
    require(type(actual) is type(expected) and actual == expected, label)
    if type(actual) in (dict, list, tuple):
        require(encoded(actual) == encoded(expected), label + ' complete literal encoding')


def ready():

        require(type(value) is str and len(value) == 64 and

                'ROOT_FILL unresolved or invalid ' + label)
    require(type(MANIFEST_FILE_COUNT) is int and MANIFEST_FILE_COUNT > 0,
            'ROOT_FILL unresolved or invalid MANIFEST_FILE_COUNT')


def regular(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path and not path.is_symlink()
            and path.is_file(), 'Canonical regular file ' + str(path))
    return path


def digest(path):
    h = hashlib.sha256()
    with regular(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def sealed(path, key):
    value = json.loads(regular(path).read_bytes())
    require(type(value) is dict and key in value, 'Complete selfsealed object ' + str(path))
    exact(value[key], hashlib.sha256(encoded({k: v for k, v in value.items() if k != key})).hexdigest(),
          'Complete selfseal ' + str(path))
    return value


def source_map():
    return {str(path.relative_to(ROOT)): digest(path)
            for folder in ('tectonics', 'tests', 'analysis', 'scripts')
            for path in sorted((ROOT / folder).rglob('*.py'))}


def project_modules():

def all_files(directory):

def component_files():
