"""Final923 evidence-only readback: immutable bytes, actual receipts and full ancestry.
Root must substitute the unique MANIFEST_RAW placeholder after closing P.
Run with a fresh outside-repository --output; F remains exactly four files.
"""
from pathlib import Path
import argparse, hashlib, json, runpy, subprocess, sys, time
ROOT=Path('/workspace/Habitable-Moon-Project')
P=ROOT/'analysis/geometric_native_common_cold_preparation_energy_validation'
F=ROOT/'analysis/geometric_native_common_cold_preparation_energy_final_readback'
MANIFEST_RAW='c3fe3def60cf9ef71c5d0049bcc0c3fe8cc12bd076f6a4fef329eb1142ec2363'
def enc(x):return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()
def load(p):return json.loads(Path(p).read_bytes())
def sealed(p):
    x=load(p);key='report_sha256' if 'report_sha256' in x else 'sha256'
    assert x[key]==hashlib.sha256(enc({k:v for k,v in x.items() if k!=key})).hexdigest(),p
    return x,key

def meaningful(p):return {str(x.relative_to(p)) for x in p.rglob('*') if x.is_file() and '__pycache__' not in x.parts and x.suffix!='.pyc'}
def check_manifest():
    assert len(MANIFEST_RAW)==64 and digest(P/'artifact_manifest.json')==MANIFEST_RAW
    manifest,_=sealed(P/'artifact_manifest.json');assert manifest['source_count']==923
    names=set()
    for name,row in manifest['files'].items():
        path=ROOT/name;assert digest(path)==row['sha256'] and path.stat().st_size==row['bytes'],name
        if path.is_relative_to(P):names.add(str(path.relative_to(P)))
    assert meaningful(P)==names|{'artifact_manifest.json'}
    return manifest

def check_archive_if_closed():
    if not (F/'archive.json').exists():return False
    archive,_=sealed(F/'archive.json')
    assert archive['status']=='PASS_FINAL923_READBACK_ARCHIVE_CLOSED'
    assert archive['manifest_raw_sha256']==MANIFEST_RAW and archive['source_count']==923
    assert archive['actual_reader_exit_code']==0 and archive['actual_reader_stderr_bytes']==0
    assert archive['actual_reader_stdout_equals_report'] is True
    assert archive['reader_execution_facts']['actual_exit_code']==0
    assert archive['reader_execution_facts']['actual_stderr_bytes']==0
    assert meaningful(F)=={'readback_source.txt','readback.json','console.log','archive.json'}
    for name,row in archive['files'].items():
        path=F/name;assert digest(path)==row['sha256'] and path.stat().st_size==row['bytes']
    assert (F/'readback.json').read_bytes()==(F/'console.log').read_bytes()
    return True

def run(output):
    start=time.perf_counter();output=Path(output).resolve()
    assert output.is_relative_to('/workspace') and not output.is_relative_to(ROOT)
    assert not output.exists();manifest=check_manifest();closed=check_archive_if_closed()
    # The pinned closure helper exposes byte-only validators; no tests, tensor integrals,
    # force solve or scientific module import is executed by this readback.
    helper=runpy.run_path(str(P/'close_evidence_source.txt'))
    old=helper['verify_old']();sources=helper['actual_sources']()
    assert sources==load(P/'frozen_source_map_923.json') and len(sources)==923
    assert set(sources)-set(old)==set(helper['NEW']) and all(sources[k]==v for k,v in old.items())
    config=load(P/'closure_facts.json')
    actual={name:helper['validation_receipt'](name,item) for name,item in config['validations'].items()}
    report,_=sealed(P/'report.json')
    assert report['status']=='PASS_FROZEN923_WHOLE_COLD_CONSTITUTIVE_PREPARATION_WORK_GATE'
    assert report['validation_receipts']==actual and report['source_count']==923
    assert report['tests_passed']==sum(x.get('tests_passed',0) for x in actual.values())
    assert report['cold_constitutive_preparation_work_computed'] is True
    assert report['reference_preparation_work_paid'] is False
    assert report['full_system_preparation_work_certified'] is False
    assert report['new_mixed_force_equilibrium_certified'] is False
    assert report['original_point_1_completed'] is False and report['original_point_1_remaining_work_blocks']==1
    assert report['accepted_new_native_intervals']==0 and report['clocks_advanced'] is False
    docs=load(P/'document_facts.json')
    assert docs['documents_finalized_before_readback'] is True
    for name,pin in docs['document_raw_sha256'].items():assert digest(name)==pin,name
    historical=P/'baseline919/historical919_handoff.txt'
    assert digest(historical)==docs['historical919_handoff_raw_sha256']
    assert (ROOT/'CLOUD_HANDOFF.md').read_bytes().endswith(historical.read_bytes())
    associations=load(P/'baseline_associations.json');assert len(associations)==10
    for target,row in associations.items():
        assert digest(ROOT/row['copy'])==row['sha256'] and digest(target)==row['sha256'],target
    inherited=ROOT/'analysis/geometric_native_common_cold_variational_work_validation/baseline_restore_source.txt'
    assert digest(inherited)==manifest['inherited919_restorer_raw_sha256']==report['inherited919_restorer_raw_sha256']
    assert report['required_baseline_target_count']==manifest['required_baseline_target_count']==90
    guard_output=Path(str(output)+'.ancestry.json');assert not guard_output.exists()
    child=subprocess.run([sys.executable,str(P/'preservation_source.txt'),'--added','--output',str(guard_output)],capture_output=True)
    assert child.returncode==0 and child.stderr==b'',(child.returncode,child.stderr)
    guard,_=sealed(guard_output)
    assert guard['status']=='PASS_CLOSED919_AND_ANCESTRY_PRESERVED'
    assert guard['source_count']==923 and guard['dependency_count']==1905
    assert guard['old_math_executed'] is False and guard['source_sha256']==sources
    assert json.loads(child.stdout)=={k:v for k,v in guard.items() if k not in ('dependency_raw_sha256','source_sha256')}
    check_manifest();check_archive_if_closed()
    for name,pin in docs['document_raw_sha256'].items():assert digest(name)==pin
    result=dict(status='PASS_FINAL923_EVIDENCE_READBACK_ONLY',manifest_raw_sha256=MANIFEST_RAW,
        manifest_file_count=len(manifest['files']),source_count=923,tests_passed=report['tests_passed'],
        old919_and_ancestry_preserved=True,old_dependency_count=1905,required_baseline_target_count=90,
        guard_actual_exit_code=child.returncode,guard_stderr_bytes=len(child.stderr),
        guard_stdout_matches_condensed_receipt=True,guard_receipt_raw_sha256=digest(guard_output),
        closed_final_archive_verified=closed,document_raw_sha256=docs['document_raw_sha256'],
        historical919_handoff_preserved=True,old_physical_math_executed=False,
        cold_constitutive_preparation_work_computed=True,reference_preparation_work_paid=False,
        full_system_preparation_work_certified=False,new_equilibrium_certified=False,
        whole_clock_all_pair_admission_authorized=False,full_system_temporal_path_work_certified=False,
        clocks_advanced=False,accepted_new_intervals=0,original_point_1_completed=False,
        original_point_1_remaining_work_blocks=1,remaining_physical_conditions=report['remaining_physical_conditions'],
        seconds=time.perf_counter()-start)
    result['sha256']=hashlib.sha256(enc(result)).hexdigest()
    with output.open('xb') as f:f.write(enc(result)+b'\n')
    print(enc(result).decode())

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--output',required=True);run(p.parse_args().output)
