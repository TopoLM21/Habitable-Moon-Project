def literal(x):

def square(x,I):return I(0 if x.lo<=0<=x.hi else min(x.lo*x.lo,x.hi*x.hi),max(x.lo*x.lo,x.hi*x.hi))
def absolute(x,I):return I(0 if x.lo<=0<=x.hi else min(abs(x.lo),abs(x.hi)),max(abs(x.lo),abs(x.hi)))

def bernstein_point(coeff,z):
 n=len(coeff)-1
 return sum((b*comb(n,k)*z**k*(1-z)**(n-k) for k,b in enumerate(coeff)),F(0))
def bernstein_cdf(coeff,z):
 n=len(coeff)-1;out=F(0)
 for k,b in enumerate(coeff):
  for j in range(n-k+1):out+=b*comb(n,k)*comb(n-k,j)*(-1)**j*z**(k+j+1)/(k+j+1)
 return out

def pi_bounds(I):
 threshold=F(1,1<<768)
 def atan_inv(q):
  x=F(1,q);power=x;total=x;n=0
  while True:
   following=-power*x*x*F(2*n+1,2*n+3)
   if abs(following)<threshold:return I(min(total,total+following),max(total,total+following))
   total+=following;power=following;n+=1
 return 16*atan_inv(5)-4*atan_inv(239)

def profile_bounds(coeff,z,I):
 if len(set(coeff))==1:return I(coeff[0])
 n=len(coeff)-1
 return sum((b*comb(n,k)*I(z.lo**k,z.hi**k)*I((1-z.hi)**(n-k),(1-z.lo)**(n-k)) for k,b in enumerate(coeff)),I(0))
