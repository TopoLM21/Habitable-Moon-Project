def encoded(x):return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def rawhash(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as stream:
  while block:=stream.read(1<<20):h.update(block)
 return h.hexdigest()
def pair(x):
 assert type(x) is F
 return [x.numerator,x.denominator]
def fresh_outside(path):
 path=Path(path).resolve()
 assert path.is_relative_to('/workspace') and not path.is_relative_to(ROOT) and not path.exists()
 return path

 started=time.monotonic();output=fresh_outside(output)
 progress=fresh_outside(progress) if progress else None
 if progress:progress.open('xb').close()
 def phase(name,**fields):
  if progress:
   with progress.open('ab') as stream:stream.write(encoded(dict(phase=name,seconds=time.monotonic()-started,**fields))+b'\n')
 phase('authentic_source_read')

  ('lambda1_explicit_illustrative_2e6_seconds',F(1),F(2*10**6),F(0)),
  ('lambda1_necessary_duration_boundary',F(1),Tmin,F(0)),
  ('lambda1_twice_necessary_duration',F(1),2*Tmin,F(0)),
  ('lambda_half_explicit_illustrative_1e6_seconds',F(1,2),F(10**6),F(0)),

  assert power.native_common_basal_work_bound_response_to_dict(power.native_common_basal_work_bound_response_from_dict(
   workdto,system=p,expected_duration_s=duration))==workdto
  assert work.maximum_net_work_j==response.maximum_net_basal_work_j
  assert not response.necessary_condition_is_sufficient and not response.actual_equilibrium_or_preparation_path_computed

   assumed_initial_kinetic_release_upper_j=pair(budget),response=dto,basal_work_response=workdto,
