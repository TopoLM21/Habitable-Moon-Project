def enc(x):
    return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()
def load(p):return json.loads(Path(p).read_bytes())
def sealed(p):
    x=load(p);key='report_sha256' if 'report_sha256' in x else 'sha256'
    assert x[key]==hashlib.sha256(enc({k:v for k,v in x.items() if k!=key})).hexdigest(),p
    return x,key

def meaningful(p):
    return {str(x.relative_to(p)) for x in p.rglob('*') if x.is_file() and '__pycache__' not in x.parts and x.suffix!='.pyc'}
def row(p):return dict(sha256=digest(p),bytes=Path(p).stat().st_size)
def write_new(p,x):
    with Path(p).open('xb') as f:f.write(enc(x)+b'\n')
def pfile(name):
    p=(P/name).resolve();assert p.is_relative_to(P) and p.is_file(),name;return p

PYTEST_LAUNCH = 'import sys;sys.path.append("/opt/codex/runtimes/codex-primary-runtime/dependencies/python/lib/python3.12/site-packages");import pytest;raise SystemExit(pytest.main(["-q","tests/test_geometric_native_common_preparation_feasibility.py","tests/test_geometric_native_common_basal_work_bound.py"]))'

def command_pairs(args, allowed, required):
    assert len(args) % 2 == 0 and all(isinstance(v,str) for v in args), 'command argument pairs'
    result = {}
    for key,value in zip(args[::2], args[1::2]):
        assert key in allowed and key not in result, ('command option',key)
        result[key] = value
    assert set(required).issubset(result)
    return result

def validate_command(name,item,facts):
    kind = item['kind'];command = facts['command']
    assert isinstance(command,list) and len(command)>=2 and all(isinstance(v,str) and v for v in command),(name,'actual command')
    assert re.fullmatch(r'python(?:3(?:\.[0-9]+)?)?',Path(command[0]).name),(name,'actual Python interpreter')
    if kind=='tests':
        assert command[1:]==['-m','pytest','-q']+TEST_TARGETS or command[1:]==['-c',PYTEST_LAUNCH],(name,'actual pytest invocation')
        return
    executed=(ROOT/command[1]).resolve()
    if kind=='native':
        assert executed==ROOT/CONTROL and CONTROL in facts['source_before_sha256'],(name,'actual native control invocation')
        args=command_pairs(command[2:],{'--output','--progress'},{'--output'})
        assert args['--output']==item['report_origin_path'],(name,'native output origin')
        if '--progress' in args:
            progress=Path(args['--progress']);assert progress.is_absolute() and progress.is_relative_to('/workspace') and not progress.is_relative_to(ROOT)
    elif kind=='audit':
        assert executed==Path(facts['executed_source_path']).resolve(),(name,'actual auditor invocation')
        args=command_pairs(command[2:],{'--native','--output','--progress'},{'--native','--output'})
        if '--progress' in args:
            progress=Path(args['--progress']);assert progress.is_absolute() and progress.is_relative_to('/workspace') and not progress.is_relative_to(ROOT)
        assert args['--native']==item['native_input_origin_path'] and args['--output']==item['report_origin_path'],(name,'actual audit inputs/output')
    elif kind in ('before','after'):
        assert executed==Path(facts['executed_source_path']).resolve(),(name,'actual preservation invocation')
        allowed={'--output','--baseline','--baseline-console','--baseline-stderr','--new-config'}
        args=command_pairs(command[2:],allowed,{'--output','--baseline','--baseline-console','--baseline-stderr'})
        assert args['--output']==facts['report_path'],(name,'actual preservation output')
        assert ('--new-config' in args)==(kind=='after'),(name,'before/after invocation')
    else:raise AssertionError(('unsupported captured invocation',kind))

def capture_shape(name,item,facts):
    assert isinstance(facts['command'],list) and facts['command'],(name,'captured command')
    assert type(facts['actual_exit_code']) is int and facts['actual_exit_code']==0,(name,'actual subprocess exit')
    for key in ('actual_stdout_raw_sha256','actual_stderr_raw_sha256','executed_capture_source_raw_sha256'):
        assert isinstance(facts[key],str) and re.fullmatch('[0-9a-f]{64}',facts[key]),(name,'mandatory raw pin',key)
    assert type(facts['actual_stderr_bytes']) is int and facts['actual_stderr_bytes']==0,(name,'captured stderr bytes')
    wrapper=Path(facts['executed_capture_source_path']).resolve()
    if 'executed_capture_source_snapshot' in item:
        assert digest(pfile(item['executed_capture_source_snapshot']))==facts['executed_capture_source_raw_sha256'],(name,'capture wrapper snapshot')
        if wrapper.exists():assert digest(wrapper)==facts['executed_capture_source_raw_sha256'],(name,'existing capture wrapper changed')
    else:assert digest(wrapper)==facts['executed_capture_source_raw_sha256'],(name,'actual capture wrapper')
    if item['kind'] in ('native','tests','audit'):
        for key in ('source_before_sha256','source_after_sha256','source_snapshots'):
            assert isinstance(facts[key],dict),(name,'mandatory scientific source map',key)
        assert set(facts['source_snapshots'])==set(facts['source_before_sha256']),(name,'complete captured source snapshot catalogue')
        assert set(NEW).issubset(facts['source_before_sha256']),(name,'four executing new sources')
        if item['kind'] in ('native','tests'):
            assert set(facts['source_before_sha256'])==set(NEW)|{CONTROL},(name,'exact science capture catalogue')
        assert type(facts['actual_child_exit_code']) is int and facts['actual_child_exit_code']==0,(name,'actual science child exit')
    if item['kind'] in ('audit','before','after'):
        assert facts['executed_source_after_raw_sha256']==facts['executed_source_raw_sha256'],(name,'executed source stability')
    validate_command(name,item,facts)

def validate_executing_sources(name,item,facts):
    """Keep current repo pins mandatory; replay outside watches from exact copies."""
    if 'executed_source_after_raw_sha256' in facts:
        assert facts['executed_source_after_raw_sha256']==facts['executed_source_raw_sha256'],(name,'executed helper source changed')
    watched=item.get('watched_source_snapshots',{})
    assert isinstance(watched,dict)
    if 'source_before_sha256' in facts:
        before=facts['source_before_sha256']
        assert isinstance(before,dict) and before==facts['source_after_sha256'],(name,'executing sources changed')
        assert set(watched).issubset(before),(name,'uncaptured watched source')
        for source,pin in before.items():
            current=(ROOT/source).resolve()
            if current.is_relative_to(ROOT):
                assert source not in watched,(name,'repo source cannot be replaced by a snapshot',source)
                assert digest(current)==pin,(name,source)
            elif source in watched:
                assert Path(source).is_absolute() and current.is_relative_to('/workspace'),(name,'outside watched source path',source)
                assert digest(pfile(watched[source]))==pin,(name,'outside watched source snapshot',source)
                if current.exists():assert digest(current)==pin,(name,'existing outside watched source changed',source)
            else:
                assert digest(current)==pin,(name,source)
    else:
        assert not watched,(name,'watched snapshots require captured before/after maps')
    for field in ('source_sha256','executed_core_source_sha256'):
        if isinstance(facts.get(field),dict):
            for source,pin in facts[field].items():assert digest(ROOT/source)==pin,(name,field,source)
    if 'executed_source_snapshot' in item:
        assert 'executed_source_raw_sha256' in facts,(name,'executed source pin missing')
        snapshot=pfile(item['executed_source_snapshot'])
        assert digest(snapshot)==facts['executed_source_raw_sha256'],(name,'executed source snapshot')
    elif 'executed_source_raw_sha256' in facts and 'executed_source_path' in facts:
        assert digest(facts['executed_source_path'])==facts['executed_source_raw_sha256'],(name,'executed source')

def validation_receipt(name,item):
    assert item['kind'] in ('tests','native','audit','before','after')
    console=pfile(item['console']);stderr=pfile(item['stderr']);facts_path=pfile(item['execution_facts'])
    facts,_=sealed(facts_path)
    capture_shape(name,item,facts)
    validate_executing_sources(name,item,facts)
    for field,path in (('console_sha256',console),('stderr_sha256',stderr),
            ('stdout_sha256',console),
            ('actual_stdout_raw_sha256',console),('actual_stderr_raw_sha256',stderr),
            ('actual_stdout_sha256',console),('actual_stderr_sha256',stderr)):
        if field in facts:assert facts[field]==digest(path),(name,field)
    for field in ('actual_stderr_bytes','stderr_bytes'):
        if field in facts:assert facts[field]==0,(name,field)
    assert item['exit_code_key'] in ('actual_exit_code','actual_child_exit_code','actual_tool_exit_code','actual_reader_exit_code')
    assert type(facts[item['exit_code_key']]) is int and facts[item['exit_code_key']]==0,(name,'actual exit')
    assert stderr.stat().st_size==0,(name,'stderr')
    # Exit evidence remains verbatim in the retained facts file; no inferred exit code.
    if item['kind']=='tests':
        count=item['tests_passed'];assert isinstance(count,int) and count>0
        text=console.read_text();assert re.search(r'\b'+str(count)+r' passed\b',text),(name,'pytest summary')
        assert not re.search(r'\b[1-9][0-9]* (?:failed|error|errors)\b',text),(name,'pytest failure')
        extra={}
        if 'report' in item:
            x,key=sealed(pfile(item['report']))
            assert x['status']==item['expected_status'] and x['tests_passed']==count
            if 'execution_facts_sha256' in x:assert x['execution_facts_sha256']==digest(facts_path)
            extra=dict(status=x['status'],report_self_sha256=x[key],report_raw_sha256=digest(pfile(item['report'])))
        return dict(kind='tests',tests_passed=count,actual_exit_code=0,
            execution_facts_raw_sha256=digest(facts_path),stdout_raw_sha256=digest(console),stderr_bytes=0,**extra)
    if item['kind'] in ('before','after'):
        x,key=guard_receipt(item,935 if item['kind']=='before' else 939)
    else:
        x,key=sealed(pfile(item['report']))
        assert console.read_bytes()==pfile(item['report']).read_bytes(),(name,'stdout equals report')
        assert x['status']==item['expected_status'],(name,x['status'])
        if item['kind']=='audit':
            assert x['native_input_path']==item['native_input_origin_path'],(name,'audited native origin')
            assertions=item.get('report_assertions',{})
            assert isinstance(assertions,dict) and assertions,(name,'mandatory independent report scope assertions')
            for path,expected in assertions.items():
                value=claim_at(x,path)
                assert type(value) is type(expected) and value==expected,(name,'independent scope assertion',path,value,expected)
    return dict(kind=item['kind'],status=x['status'],report_self_sha256=x[key],
        report_raw_sha256=digest(pfile(item['report'])),actual_exit_code=0,
        execution_facts_raw_sha256=digest(facts_path),stdout_raw_sha256=digest(console),stderr_bytes=0)

def claim_at(native, path):
    assert isinstance(path, str) and path
    value = native
    for part in path.split('.'):
        value = value[int(part)] if isinstance(value, list) else value[part]
    return value

def extract_science(native, config):
    mapping = config.get('native_claims_keys', {})
    assert set(mapping).issubset(set(REQUIRED_SCIENCE) | {'remaining_physical_conditions'})
    result = {}
    for name, expected in REQUIRED_SCIENCE.items():
        value = claim_at(native, mapping.get(name, name))
        assert type(value) is type(expected) and value == expected, (name, value, expected)
        result[name] = value
    conditions = claim_at(native, mapping.get('remaining_physical_conditions', 'remaining_physical_conditions'))
    assert isinstance(conditions, list) and len(conditions) == 3
    assert all(isinstance(v, str) and v.strip() for v in conditions)
    result['remaining_physical_conditions'] = conditions
    assertions = config.get('native_assertions', {})
    assert isinstance(assertions, dict)
    for path, expected in assertions.items():
        value = claim_at(native, path)
        assert type(value) is type(expected) and value == expected, (path, value, expected)
    result['additional_actual_native_assertions'] = assertions
    return result

def seal_final(facts_path):
    facts,_=sealed(facts_path)
    assert type(facts['actual_exit_code']) is int and facts['actual_exit_code']==0
    assert type(facts['actual_stderr_bytes']) is int and facts['actual_stderr_bytes']==0
    report_path=Path(facts['report_path']);stdout=Path(facts['actual_stdout_path']);stderr=Path(facts['actual_stderr_path'])
    assert digest(stdout)==facts['actual_stdout_raw_sha256'] and digest(stderr)==facts['actual_stderr_raw_sha256']
    assert digest(report_path)==facts['report_raw_sha256'] and facts['actual_stdout_equals_report'] is True
    assert stderr.stat().st_size==0 and report_path.read_bytes()==stdout.read_bytes()
    assert facts['executed_source_raw_sha256']==facts['executed_source_after_raw_sha256']==digest(F/'readback_source.txt')
    command=facts['command']
    assert isinstance(command,list) and len(command)==4 and re.fullmatch(r'python(?:3(?:\.[0-9]+)?)?',Path(command[0]).name)
    assert (ROOT/command[1]).resolve()==Path(facts['executed_source_path']).resolve()
    assert command[2:]==['--output',str(report_path)]
    wrapper=Path(facts['executed_capture_source_path'])
    assert digest(wrapper)==facts['executed_capture_source_raw_sha256']
    result,_=sealed(report_path);assert result['status']=='PASS_FINAL939_EVIDENCE_READBACK_ONLY'
    assert result['manifest_raw_sha256']==digest(P/'artifact_manifest.json')
    assert meaningful(F)=={'readback_source.txt','readback.json','console.log'}
    assert (F/'readback.json').read_bytes()==report_path.read_bytes() and (F/'console.log').read_bytes()==stdout.read_bytes()
    if 'executed_source_raw_sha256' in facts:assert digest(F/'readback_source.txt')==facts['executed_source_raw_sha256']
    archive=dict(status='PASS_FINAL939_READBACK_ARCHIVE_CLOSED',source_count=939,
        files={n:row(F/n) for n in ('readback_source.txt','readback.json','console.log')},
        manifest_raw_sha256=digest(P/'artifact_manifest.json'),actual_reader_exit_code=0,
        actual_reader_stderr_bytes=0,actual_reader_stdout_equals_report=True,
        reader_execution_facts_raw_sha256=digest(facts_path),reader_execution_facts=facts,point_1_completed=False)
    archive['sha256']=hashlib.sha256(enc(archive)).hexdigest();write_new(F/'archive.json',archive)
    assert meaningful(F)=={'readback_source.txt','readback.json','console.log','archive.json'}
    print(enc(dict(status='PASS_CLOSED939_FOUR_FILE_ARCHIVE',archive_raw_sha256=digest(F/'archive.json'))).decode())
