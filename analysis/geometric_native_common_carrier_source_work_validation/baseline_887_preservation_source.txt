def primitives():
    """Authenticate/define an existing stdlib guard without invoking its main."""
    helper = PARENT / 'baseline_883_preservation_source.txt'
    if hashlib.sha256(helper.read_bytes()).hexdigest() != PRESERVATION883_RAW:
        raise ValueError('Immutable stdlib883 primitive source changed')
    module = types.ModuleType('_private_891_preservation_primitives')
    module.__file__ = str(helper)
    exec(compile(helper.read_bytes(), str(helper), 'exec'), module.__dict__)
    return module


def project_modules():
    return {name: module for name, module in sys.modules.items()
            if name in ('tectonics', 'tests', 'analysis') or
            name.startswith(('tectonics.', 'tests.', 'analysis.'))}

def run(additional_source_sha256, *, output_path):

    started = time.perf_counter()
    modules_before = project_modules()
    b = primitives()
    b.require(type(additional_source_sha256) is dict, 'Literal additional four source map')
    b.exact(set(additional_source_sha256), set(ADDED_NAMES), 'Exactly four authorized891 source paths')
    requested = Path(output_path)
    b.require(requested.is_absolute() and requested.resolve() == requested and
              not requested.exists() and not requested.is_symlink() and
              requested.parent.is_dir() and requested.parent.resolve() == requested.parent and
              requested.suffix == '.json' and not requested.is_relative_to(ROOT) and
              (requested.is_relative_to(Path('/workspace')) or requested.is_relative_to(Path('/tmp'))),
              'Fresh canonical exclusive JSON output outside repository')
    producer = b.regular(Path(__file__).resolve())
    producer_raw = b.digest(producer)
    frozen887, baseline, pins = preserved_inputs(b)
    declared = dict(frozen887)
    for name, expected in additional_source_sha256.items():
        b.require(name not in frozen887, 'Authorized891 source is genuinely new')
        b.pin(pins, ROOT / name, expected)
        declared[name] = expected
    b.exact(b.source_map(), declared, 'Actual891 equals immutable887 plus exactly frozen four')
    b.pin(pins, producer, producer_raw)
    b.require(str(requested) not in pins, 'Fresh output disjoint from all dependencies')
    after_frozen, after_baseline, after_pins = preserved_inputs(b)
    b.exact(after_frozen, frozen887, 'Saved887 source authority unchanged AFTER')
    b.exact(after_baseline, baseline, 'Saved pre-new-source887 receipt unchanged AFTER')
    for name, expected in pins.items():
        if name in after_pins:
            b.exact(after_pins[name], expected, 'Raw dependency unchanged AFTER ' + name)
        else:
            b.exact(b.digest(name), expected, 'Added source/producer unchanged AFTER ' + name)
    b.exact(b.source_map(), declared, 'Complete actual891 source map unchanged AFTER')
    modules_after = project_modules()
    b.exact(set(modules_after), set(modules_before), 'Real caller project module names preserved')
    b.require(all(modules_after[name] is module for name, module in modules_before.items()),
              'Real caller project module identities preserved')
    result = dict(
        format='native-common-carrier-source-work-prior887-preservation-1',
        status='PASS_IMMUTABLE887_WITH_EXACT_FOUR_DECLARED_SOURCE_ADDITIONS',
        prior_python_source_count=887, current_python_source_count=891,
        prior_python_source_map_canonical_sha256=b.value_digest(frozen887),
        current_python_source_map_canonical_sha256=b.value_digest(declared),

        all887_python_sources_unchanged=True, complete_actual_source_map_checked_before_after=True,
        prior_887_metadata_is_authenticated_saved_readback_only=True,

        fresh_pre_new_source_887_readback_raw_sha256=FRESH_REPORT_RAW,

        fresh_pre_new_source_887_readback_seconds=baseline['seconds'],

        recorded_raw_dependency_pin_count=len(pins), dependency_raw_sha256=pins,

        preserved_inherited879_final_readback_archive_file_count=6,
        authoritative_ancestry_release_closure_count=11,

        preserved_current887_release_closed_file_count=5,
        temporary_dependency_copy_association_count=15,
