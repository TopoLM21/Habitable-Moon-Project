def power_two(v):return F(1<<v)if v>=0 else F(1,1<<-v)
def floor_log_two(v):
 n,d=v.numerator,v.denominator;e=n.bit_length()-d.bit_length()

def rounded(v,precision=96):

 bits=(3322*(precision+16)+999)//1000+8

def encode_scalar(v):return [v.numerator,v.denominator]
