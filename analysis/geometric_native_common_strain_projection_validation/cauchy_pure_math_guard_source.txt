class ExpectedError:
    def __init__(self, kind, match):
        self.kind, self.match = kind, match
    def __enter__(self):
        return self
    def __exit__(self, kind, value, trace):
        assert kind is not None and issubclass(kind, self.kind)
        assert re.search(self.match, str(value))
        return True


def select(path, names, namespace):
    data = path.read_bytes()
    parsed = ast.parse(data)
    nodes = [node for node in parsed.body
             if isinstance(node, (ast.FunctionDef, ast.ClassDef)) and node.name in names]
    assert {node.name for node in nodes} == names
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(path), 'exec'), namespace)
    return hashlib.sha256(data).hexdigest()

# Independently certify upward and minimal dyadic square-root bounds over
# exact irrational, perfect-square, zero, tiny and large arguments.
I = namespace['I']
ratios = (F(0), F(1, 4), F(1), F(2), F(1, 3), F(7, 13),
          F(10**20, 3), F(1, 10**200))
cap_checks = 0
for precision in (64, 96, 128, 512):
    bits = (3322*(precision+16)+999)//1000+8
    step = F(1, 1 << bits)
    for ratio in ratios:
        cap = namespace['_projection_cauchy_cap'](I(1), I(ratio), precision)
        assert type(cap) is F and cap >= 0 and cap*cap >= ratio
        assert cap == 0 or (cap-step)*(cap-step) < ratio
        assert cap*step.denominator == int(cap*step.denominator)
        cap_checks += 1

# The cap bounds the unchanged quotient, never its derivatives.

zi = (I(0),)*7
zh = tuple(zi for _ in range(7))

    I(0, 1), zi, zh, zero, 128)

assert contracted['projection_gradient_bounds'] == (I(1),)+(I(0),)*6
