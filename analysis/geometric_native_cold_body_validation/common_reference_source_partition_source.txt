The referenced stdlib curved peer source/report are separate proof dependencies.

def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
                      ensure_ascii=False, allow_nan=False).encode()


def raw_read(path, digest):
    raw = Path(path).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == digest, ('raw input changed', str(path))
    return json.loads(raw)


def pair(value):
    assert type(value) is list and len(value) == 2
    result = F(*value)
    assert [result.numerator, result.denominator] == value
    return result


def dyadic_lower(value):
    return F((value.numerator * SCALE) // value.denominator, SCALE)


def dyadic_upper(value):
    return F(-((-value.numerator * SCALE) // value.denominator), SCALE)


class I:
    def __init__(self, lo, hi=None):
        if isinstance(lo, I):
            self.lo, self.hi = lo.lo, lo.hi
        else:
            self.lo, self.hi = F(lo), F(lo if hi is None else hi)
        assert self.lo <= self.hi

    @staticmethod
    def rounded(lo, hi):
        return I(dyadic_lower(lo), dyadic_upper(hi))

    def __add__(self, other):
        other = I(other)
        return I.rounded(self.lo + other.lo, self.hi + other.hi)

    __radd__ = __add__

    def __neg__(self):
        return I(-self.hi, -self.lo)

    def __sub__(self, other):
        return self + -I(other)

    def __mul__(self, other):
        other = I(other)
        pp = (self.lo * other.lo, self.lo * other.hi,
              self.hi * other.lo, self.hi * other.hi)
        return I.rounded(min(pp), max(pp))

    __rmul__ = __mul__

    def __truediv__(self, other):
        other = I(other)
        assert not other.lo <= 0 <= other.hi
        return self * I.rounded(1 / other.hi, 1 / other.lo)

    def payload(self):
        return dict(lo=[self.lo.numerator, self.lo.denominator],
                    hi=[self.hi.numerator, self.hi.denominator])

    def abs_upper(self):
        return max(abs(self.lo), abs(self.hi))


def read_i(raw):
    return I(pair(raw['lo']), pair(raw['hi']))


def dot(a, b):
    return sum((x * y for x, y in zip(a, b)), I(0))


def cross(a, b):
    return (a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2],
            a[0]*b[1]-a[1]*b[0])


def rotate(vector, axis):
    """Full literal SO(3), exact Taylor centre plus uniform norm remainder."""
    vector, axis = tuple(map(I, vector)), tuple(map(I, axis))
    rho = sum(x.abs_upper() for x in axis)
    maximum = max(x.abs_upper() for x in vector)
    assert rho < 1
    term, result = vector, vector
    for degree in range(1, ORDER + 1):
        term = tuple(x / degree for x in cross(axis, term))
        result = tuple(x + y for x, y in zip(result, term))
    tail = maximum * rho**(ORDER+1) / factorial(ORDER+1)
    tail /= 1 - rho/F(ORDER+2)
    return tuple(x + I(-tail, tail) for x in result), tail


def step_key(raw):
    assert set(raw) == {'omega_rad_per_myr', 'dt_myr'}
    return tuple(F.from_float(x) for x in raw['omega_rad_per_myr']), F.from_float(raw['dt_myr'])


def reduce_history(raw):
    """Cancel only exact adjacent inverse increments; preserve full DTO elsewhere."""
    result = []
    for step in raw:
        axis, duration = step_key(step)
        if not any(axis) or not duration:
            continue
        if result and duration == result[-1][1] and axis == tuple(-x for x in result[-1][0]):
            result.pop()
        else:
            result.append((axis, duration))
    return tuple(result)


def canonical_planes(poly, owner_history):
    """Actual normal's ordered local+pose word, reduced by exact SO(3) identities."""
    assert poly['version'] == 'geometric-native-polygons-1'
    result = []
    for plane in poly['supports']:
        normal = plane['normal']
        assert sum(bool(x) for x in normal) == 1
        axis = next(i for i, x in enumerate(normal) if x)
        assert abs(normal[axis]) == 1
        word = reduce_history(plane['rotations'] + poly['pose'])
        owner = next((o for o, h in owner_history.items() if word == reduce_history(h)), None)
        assert not word or owner is not None, ('unexpected literal plane word', word)
        index = axis if not word else 3 + 3*owner + axis
        result.append((index, 1 if normal[axis] > 0 else -1))
    return tuple(result)


def belongs(pattern, planes):
    return all(pattern[index] == sign for index, sign in planes)


def in_pieces(pattern, pieces):
    return any(belongs(pattern, planes) for planes in pieces)


def strict_cell_witness(pattern, i, j, signed_ray, normals):
    """Rational point proved strictly feasible for its ENTIRE cell sign profile.

    This supplements the vertex enumeration with 74 explicit witnesses.  It
    solves the two incident midpoint plane equations, then chooses a strictly
    bounded perturbation which cannot change any other strict source sign.
    Midpoints are scratch coordinates only: all nine actual SO(3) plane signs
    are re-certified by their complete interval normals at the final point.
    """
    ni = tuple((x.lo+x.hi)/2 for x in normals[i])
    nj = tuple((x.lo+x.hi)/2 for x in normals[j])
    a = sum(x*x for x in ni)
    b = sum(x*x for x in nj)
    c = sum(x*y for x, y in zip(ni, nj))
    det = a*b-c*c
    assert det > 0
    si, sj = pattern[i], pattern[j]
    g = tuple(dyadic_lower(((b*si-c*sj)*x+(a*sj-c*si)*y)/det)
              for x, y in zip(ni, nj))
    r = tuple((x.lo+x.hi)/2 for x in signed_ray)
    margin = min((pattern[k]*dot(normals[k], signed_ray)).lo
                 for k in range(9) if k not in (i, j))
    assert margin > 0
    epsilon = dyadic_lower(margin/(32*(1+3*max(map(abs, g)))))
    assert epsilon > 0
    point = tuple(dyadic_lower(x+epsilon*y) for x, y in zip(r, g))
    bounds = tuple(dot(normal, tuple(I(x) for x in point)) for normal in normals)
    assert all((sign*value).lo > 0 for sign, value in zip(pattern, bounds))
    return dict(exact_rational_direction=[[x.numerator, x.denominator] for x in point],
                all9_strict_literal_plane_bounds=[x.payload() for x in bounds],
                source_vertex_planes=[i, j], perturbation_magnitude=[epsilon.numerator, epsilon.denominator],
                midpoint_point_used_as_membership_authority=False)
