@dataclass(frozen=True)
class I:
    lo: F
    hi: F=None
    def __post_init__(self):
        object.__setattr__(self,'lo',F(self.lo)); object.__setattr__(self,'hi',F(self.lo if self.hi is None else self.hi))
        assert self.lo <= self.hi
    def __add__(self,x):
        x=iv(x);return I(self.lo+x.lo,self.hi+x.hi)
    __radd__=__add__
    def __neg__(self): return I(-self.hi,-self.lo)
    def __sub__(self,x): return self+-iv(x)
    def __rsub__(self,x): return iv(x)+-self
    def __mul__(self,x):
        x=iv(x);z=[self.lo*x.lo,self.lo*x.hi,self.hi*x.lo,self.hi*x.hi];return I(min(z),max(z))
    __rmul__=__mul__
    def __truediv__(self,x):
        x=iv(x);assert not x.lo<=0<=x.hi;return self*I(1/x.hi,1/x.lo)
    def __rtruediv__(self,x): return iv(x)/self
    @property
    def mid(self): return (self.lo+self.hi)/2
    def dto(self):
        # Only compress huge derived exact fractions, always outward. The
        # authenticated source's literal area/clock/envelope rational pairs
        # remain unchanged; no process-wide integer string limit override.
        value=self
        if max(v.bit_length() for x in (self.lo,self.hi) for v in (abs(x.numerator),x.denominator))>12000:
            value=roundi(self,384)
        return {'lo':fr(value.lo),'hi':fr(value.hi)}

def iv(x):return x if isinstance(x,I) else I(x)
def readi(x):return I(rat(x['lo']),rat(x['hi']))
def power2(e):return F(2**e) if e>=0 else F(1,2**(-e))
def roundi(x,p):
    x=iv(x);a=max(abs(x.lo),abs(x.hi))
    if not a:return I(0)
    e=a.numerator.bit_length()-a.denominator.bit_length()
    if a<power2(e):e-=1
    scale=power2((3322*p+999)//1000+8-e)
    a,b=x.lo*scale,x.hi*scale
    return I(F(a.numerator//a.denominator)/scale,F(-((-b.numerator)//b.denominator))/scale)
def deriv(co,n=1):return tuple(x*F(factorial(k),factorial(k-n)) for k,x in enumerate(co) if k>=n) or (F(0),)
def cross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])
def normal_coefficients(d,r,n):
    # Independent convolution of the two matrix exponential series.
    right=[(F(-1),F(0),F(0))]
    for k in range(1,n+1):right.append(tuple(x/k for x in cross(r,right[-1])))
    out=[(F(0),)*3 for _ in range(n+1)]
    for k,row in enumerate(right):
        left=row
        for j in range(n-k+1):
            if j:left=tuple(x/j for x in cross(tuple(-x for x in d),left))
            out[k+j]=tuple(a+b for a,b in zip(out[k+j],left))
    return tuple(out)

class Poly:
    def __init__(self,co,tails,p,n):
        self.p,self.n=p,n
        ce=tuple(roundi(I(x),p+24).mid for x in co)
        er=tuple(abs(a-b) for a,b in zip(co,ce))
        self.co=ce
        self.tails=tuple(roundi(I(t+sum((x*F(factorial(k),factorial(k-j)) for k,x in enumerate(er) if k>=j),F(0))),p+24).hi for j,t in enumerate(tails))
    def derivative(self):return Poly(deriv(self.co),self.tails[1:],self.p,self.n)
    def __neg__(self):return Poly(tuple(-x for x in self.co),self.tails,self.p,self.n)
    def __sub__(self,x):
        n=max(len(self.co),len(x.co));return Poly(tuple((self.co[k] if k<len(self.co) else 0)-(x.co[k] if k<len(x.co) else 0) for k in range(n)),tuple(a+b for a,b in zip(self.tails,x.tails)),self.p,self.n)
    def __mul__(self,x):
        out=[F(0)]*(len(self.co)+len(x.co)-1)
        for i,a in enumerate(self.co):
            for j,b in enumerate(x.co):out[i+j]+=a*b
        tails=[]
        for j in range(min(len(self.tails),len(x.tails))):
            t=sum((abs(a)*F(factorial(k),factorial(k-j)) for k,a in enumerate(out) if k>self.n and k>=j),F(0))
            for k in range(j+1):
                l=j-k;ap=sum(map(abs,deriv(self.co,k)),F(0));bp=sum(map(abs,deriv(x.co,l)),F(0))
                t+=comb(j,k)*(self.tails[k]*bp+x.tails[l]*ap+self.tails[k]*x.tails[l])
            tails.append(t)
        return Poly(tuple(out[:self.n+1]),tuple(tails),self.p,self.n)
    def bounds(self,u,j=0):
        y=I(0)
        for a in reversed(deriv(self.co,j)):y=roundi(y*u+a,self.p+16)
        return roundi(y+I(-self.tails[j],self.tails[j]),self.p+16)

def bernstein(co,u,tail):
    # Exact Bernstein convex hull; independent of project clipping/pose code.
    n=len(co)-1;l,w=u.lo,u.hi-u.lo
    power=tuple(sum((co[k]*comb(k,j)*l**(k-j)*w**j for k in range(j,n+1)),F(0)) for j in range(n+1))
    be=tuple(sum((power[k]*F(comb(i,k),comb(n,k)) for k in range(i+1)),F(0)) for i in range(n+1))
    return I(min(be)-tail,max(be)+tail)
def factored_bounds(co,u,rate,j=0):
    n=len(co)-1
    tail=rate**(n+2)/factorial(n+2)*F(factorial(n+1),factorial(n+1-j))/(1-rate/F(n+2-j))
    return bernstein(deriv(co,j),u,tail)
def rotation(a,n=24):
    rate=sum(map(abs,a),F(0));tail=rate**(n+1)/factorial(n+1)/(1-rate/F(n+2))
    cols=[]
    for j in range(3):
        term=tuple(F(i==j) for i in range(3));total=term
        for k in range(1,n+1):
            term=tuple(x/k for x in cross(a,term));total=tuple(x+y for x,y in zip(total,term))
        cols.append(tuple(I(x-tail,x+tail) for x in total))
    return tuple(tuple(cols[j][i] for j in range(3)) for i in range(3))
