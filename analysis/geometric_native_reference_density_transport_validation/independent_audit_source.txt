def enc(x):return json.dumps(x,sort_keys=True,separators=(',',':')).encode()
def tensor(x,depth):
 if depth==0:return F(*x)
 return tuple(tensor(v,depth-1) for v in x)
def inverse(A):
 n=len(A);a=[list(row)+[F(i==j) for j in range(n)] for i,row in enumerate(A)]
 determinant=F(1)
 for i in range(n):
  pivot=next(k for k in range(i,n) if a[k][i])
  if pivot!=i:a[i],a[pivot]=a[pivot],a[i];determinant=-determinant
  v=a[i][i];determinant*=v;a[i]=[x/v for x in a[i]]
  for k in range(n):
   if k!=i:
    v=a[k][i];a[k]=[x-v*y for x,y in zip(a[k],a[i])]
 return determinant,tuple(tuple(row[n:]) for row in a)
def mul(A,B):return tuple(tuple(sum((a*b for a,b in zip(row,col)),F(0)) for col in zip(*B)) for row in A)
def transpose(A):return tuple(zip(*A))
