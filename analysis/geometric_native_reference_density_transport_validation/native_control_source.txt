from pathlib import Path
from fractions import Fraction as F
import sys,json,hashlib,time,argparse
ROOT=Path(__file__).resolve().parents[2];sys.path.insert(0,str(ROOT))
from tectonics.geometric_native_reference_density_transport import ReferenceDensityTwoJet,transport_reference_density,pullback_density_to_modes

def encode(x):
 if type(x) is F:return [x.numerator,x.denominator]
 if isinstance(x,(tuple,list)):return [encode(v) for v in x]
 if isinstance(x,dict):return {k:encode(v) for k,v in x.items()}
 return x
def enc(x):return json.dumps(x,sort_keys=True,separators=(',',':')).encode()
def main(output):
 t=time.perf_counter();n=11
 H=tuple(tuple(F((a+1)*(b+1),113)+F(3 if a==b else 0) for b in range(n)) for a in range(n))
 density=ReferenceDensityTwoJet(F(987654321,17),tuple(F(7*a-19,23) for a in range(n)),H)
 Fold=tuple(tuple(F(x) for x in row) for row in ((1,F(1,5),0),(0,1,F(1,7)),(F(1,11),0,1)))
 bridges=(((2,1,0),(0,3,1),(1,0,2)),((1,2,0),(0,1,3),(0,0,1)),((2,0,0),(0,3,0),(0,0,4)))
 rows=[]
 for rowno,raw in enumerate(bridges):
  B=tuple(tuple(F(v) for v in r) for r in raw);d=100 if rowno==0 else 6
  D=tuple(tuple(F((a+1)*(i+2)-5,31) for i in range(d)) for a in range(n))
  Q=tuple(tuple(tuple(F((a+2)*(i+j+1),37) for j in range(d)) for i in range(d)) for a in range(n))
  out=transport_reference_density(B,Fold,density)
  Dnew=[];Qnew=[]
  for a in range(n):
   if a>=9:Dnew.append(D[a]);Qnew.append(Q[a]);continue
   r,c=divmod(a,3)
   Dnew.append(tuple(sum((D[3*r+k][i]*out.inverse_bridge[k][c] for k in range(3)),F(0)) for i in range(d)))
   Qnew.append(tuple(tuple(sum((Q[3*r+k][i][j]*out.inverse_bridge[k][c] for k in range(3)),F(0)) for j in range(d)) for i in range(d)))
  original=pullback_density_to_modes(density,D,Q,F(17))
  changed=pullback_density_to_modes(out.new_density,tuple(Dnew),tuple(Qnew),F(17)*out.reference_volume_ratio)
  assert original==changed
  rows.append(encode(dict(bridge=B,old_deformation=Fold,old_energy=density.energy_density,old_gradient=density.gradient,old_hessian=H,mode_count=d,first=D,second=Q,inverse=out.inverse_bridge,volume_ratio=out.reference_volume_ratio,new_deformation=out.new_deformation_gradient,new_energy=out.new_density.energy_density,new_gradient=out.new_density.gradient,new_hessian=out.new_density.hessian,original_modal=original,transported_modal=changed)))
 result=dict(status='PASS_REFERENCE_DENSITY_COVARIANCE_CONTROL',cases=rows,case_count=len(rows),hundred_mode_hessian_verified=True,physical_preparation_work_paid=False,reference_gap_resolved=False,new_equilibrium_certified=False,accepted_new_intervals=0,seconds=time.perf_counter()-t)
 result['sha256']=hashlib.sha256(enc(result)).hexdigest()
 with Path(output).open('xb') as f:f.write(enc(result)+b'\n')
 print(json.dumps({k:v for k,v in result.items() if k!='cases'},sort_keys=True))
if __name__=='__main__':
 p=argparse.ArgumentParser();p.add_argument('--output',required=True);main(p.parse_args().output)
