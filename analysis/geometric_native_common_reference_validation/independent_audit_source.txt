def literal_word(rows):
 result=[]
 for row in rows:
  angle=tuple(F.from_float(float(w))*F.from_float(float(row['dt_myr'])) for w in row['omega_rad_per_myr'])
  if not any(angle):continue
  if result and result[-1]==tuple(-x for x in angle):result.pop()
  else:result.append(angle)
 return tuple(result)

def supports(poly,histories):
