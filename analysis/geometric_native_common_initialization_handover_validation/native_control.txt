def encoded(x):return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def digest(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as stream:
  for block in iter(lambda:stream.read(1<<20),b''):h.update(block)
 return h.hexdigest()
def pair(x):return [x.numerator,x.denominator]

def run(output,progress=None):
 from tectonics.geometric_native_common_initialization_source import (
  bind_native_common_initialization_source,
  native_common_initialization_source_evidence,
  native_common_initialization_source_to_dict,
  native_common_initialization_source_from_dict,
 )
 from tectonics.geometric_native_common_initialization_handover import (
  bind_native_common_initialization_handover,
  evaluate_native_common_initialization_handover,
  native_common_initialization_handover_to_dict,
  native_common_initialization_handover_from_dict,
  native_common_initialization_handover_response_to_dict,
  native_common_initialization_handover_response_from_dict,
  native_common_initialization_handover_receipt,
 )
 started=time.monotonic();output=Path(output).resolve()
 assert output.is_relative_to('/workspace') and not output.is_relative_to(ROOT) and not output.exists()
 if progress is not None:
  progress=Path(progress).resolve()
  assert progress.is_relative_to('/workspace') and not progress.is_relative_to(ROOT) and not progress.exists()
 def phase(label,**extra):
  if progress is not None:
   with progress.open('a') as stream:stream.write(encoded(dict(phase=label,seconds=time.monotonic()-started,**extra)).decode()+'\n')
 phase('started')
 authorities={name:digest(ROOT/path) for name,path in SOURCES.items()}
 for name,raw in EXPECTED_RAW.items():assert authorities[name]==raw,(name,'original raw authority')
 phase('authorities_hashed')
 packets=[json.loads((ROOT/SOURCES[name]).read_bytes()) for name in ('source935','source867','source907','source895')]
 phase('full_packets_loaded')
 context=bind_native_common_initialization_source(*packets,provenance='NEW943 full accepted-source provenance and frozen whole Cartesian references')
 cdto=native_common_initialization_source_to_dict(context)
 restored_context=native_common_initialization_source_from_dict(cdto,*packets)
 assert native_common_initialization_source_to_dict(restored_context)==cdto
 evidence=native_common_initialization_source_evidence(context)
 assert evidence['physical_initialization_ready'] is False
 phase('context_authenticated_and_restarted')
 del packets;gc.collect()
 system=bind_native_common_initialization_handover(context,provenance='NEW943 exact source/target geometry, work and initialization residuals without physical admission')
 sdto=native_common_initialization_handover_to_dict(system)
 restored_system=native_common_initialization_handover_from_dict(sdto,context=restored_context)
 assert native_common_initialization_handover_to_dict(restored_system)==sdto
 phase('system_restarted')
 queries=[]
 for label,increment in (
  ('formal_material_e1',(F(1),F(0),F(0))),
  ('formal_material_e2',(F(0),F(1),F(0))),
  ('formal_material_e3',(F(0),F(0),F(1))),
  ('formal_material_mixed',(F(1,2),F(-2),F(3,7))),
 ):
  response=evaluate_native_common_initialization_handover(system,virtual_material_increment_m=increment)
  dto=native_common_initialization_handover_response_to_dict(response)
  restored=native_common_initialization_handover_response_from_dict(dto,system=restored_system,expected_virtual_material_increment_m=increment)
  assert native_common_initialization_handover_response_to_dict(restored)==dto
  receipt=native_common_initialization_handover_receipt(response)
  assert len(receipt['side_geometry_receipts'])==32
  assert len(receipt['contact_force_work_receipts'])==16
  assert len(receipt['beam_energy_receipts'])==8
  assert response.cartesian_bridge_diagrams_verified is True
  assert response.covector_virtual_work_diagrams_verified is True
  assert response.received_material_embedding_preserved is False
  assert response.accepted_mechanical_initialization_compatible is False
  queries.append(dict(label=label,virtual_material_increment_m=[pair(x) for x in increment],
                      response=dto,receipt=receipt,complete_response_restart_equal=True))
  phase('query_completed',query_label=label)
 assert authorities=={name:digest(ROOT/path) for name,path in SOURCES.items()}
 closed939=json.loads((ROOT/'analysis/geometric_native_common_preparation_feasibility_validation/report.json').read_bytes())
 required=('general_finite_owner_state_chart_defined','full_mixed_force_and_tangent_certified',
 'reference_preparation_work_paid','full_system_preparation_work_certified',
 'historical_archived_three_dimensional_prestress_recovered','new_mixed_force_equilibrium_certified',
 'whole_clock_all_pair_admission_authorized','physical_material_admission_authorized',
 'full_system_temporal_path_work_certified','clocks_advanced','source_geometry_elapsed_time_s',

  source_context=cdto,handover_system=sdto,queries=queries,
  complete_source_context_restart_equal=True,complete_handover_system_restart_equal=True,

  seconds=time.monotonic()-started)
 packet['sha256']=hashlib.sha256(encoded(packet)).hexdigest()
