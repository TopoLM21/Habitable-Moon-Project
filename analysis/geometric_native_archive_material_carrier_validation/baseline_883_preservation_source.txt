def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
                      ensure_ascii=False, allow_nan=False).encode()


def value_digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()


def require(condition, label):
    if not condition:
        raise ValueError(label)


def exact(actual, expected, label):
    require(type(actual) is type(expected) and actual == expected, label)
    if type(actual) in (dict, list, tuple):
        require(encoded(actual) == encoded(expected), label + ' complete literal encoding')


def regular(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path and
            not path.is_symlink() and path.is_file(), 'Canonical regular file ' + str(path))
    return path


def digest(path):
    path = regular(path)
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def sealed(path, key):
    value = json.loads(regular(path).read_bytes())
    require(type(value) is dict and key in value, 'Complete sealed object ' + str(path))
    body = dict(value)
    exact(body.pop(key), value_digest(body), 'Complete selfseal ' + str(path))
    return value


def source_map():
    return {str(path.relative_to(ROOT)): digest(path)
            for directory in ('tectonics', 'tests', 'scripts', 'analysis')
            for path in sorted((ROOT / directory).rglob('*.py'))}


def project_modules():
    return {name: module for name, module in sys.modules.items()
            if name in ('tectonics', 'tests', 'analysis') or
            name.startswith(('tectonics.', 'tests.', 'analysis.'))}


def files_under(parent):
    return {str(path.relative_to(parent)) for path in parent.rglob('*')
            if path.is_file() and '__pycache__' not in path.parts and path.suffix != '.pyc'}


def pin(pins, path, expected):
    path = regular(path)
    require(path.is_relative_to(ROOT) or path.is_relative_to(Path('/tmp')) or
            path.is_relative_to(Path('/workspace')), 'Authorized preserved dependency')
    require(type(expected) is str and len(expected) == 64 and
            all(c in '0123456789abcdef' for c in expected), 'Raw SHA256 shape')
    name = str(path)
    if name in pins:
        exact(pins[name], expected, 'Overlapping immutable pin ' + name)
        return
    exact(digest(path), expected, 'Preserved raw bytes ' + name)
    pins[name] = expected


def lineage(reader_source):
    tree = ast.parse(reader_source)
    matches = [ast.literal_eval(node.value) for node in tree.body
               if isinstance(node, ast.Assign) and
               any(isinstance(target, ast.Name) and target.id == 'ANCESTRY'
                   for target in node.targets)]
    exact(len(matches), 1, 'Exactly one saved eleven-release declaration')
    exact(len(matches[0]), 11, 'Eleven inherited authoritative releases')
    return matches[0]


def strict_release(relative, count, tests, wrapper_raw, pins):
    directory = ROOT / relative
    report_path = directory / 'report.json'
    pin(pins, report_path, wrapper_raw)
    report = sealed(report_path, 'report_sha256')
    for key, expected in (('status', 'PASS'), ('source_count', count),
                          ('pytest_counts', {'passed': tests})):
        exact(report[key], expected, 'Saved release ' + relative + ' ' + key)
    exact(set(report['outputs']), set(OUTPUT_NAMES), 'Four saved output rows')
    # Release archives have exact five files, including any unexpected cache.
    exact({str(path.relative_to(directory)) for path in directory.rglob('*') if path.is_file()},
          set(OUTPUT_NAMES) | {'report.json'}, 'Strict five-file release ' + relative)
    for name, row in report['outputs'].items():
        exact(set(row), {'bytes', 'sha256'}, 'Complete saved output pin row')
        exact(regular(directory / name).stat().st_size, row['bytes'], 'Saved output size')
        pin(pins, directory / name, row['sha256'])

def run(additional_source_sha256, *, output_path):
    """Return/write one preservation receipt for actual883+exact declared four.

    additional_source_sha256 must contain exactly ADDED_NAMES and the frozen
    SHA256 of every added file. output_path is a fresh JSON outside the repo.
    Existing caller project modules are retained by exact name and identity.
    """
    started = time.perf_counter()
    require(type(additional_source_sha256) is dict, 'Exact declared new source map')
    exact(set(additional_source_sha256), set(ADDED_NAMES), 'Exactly four authorized new Python paths')
    output_path = Path(output_path)
    require(output_path.is_absolute() and output_path.resolve() == output_path and
            not output_path.is_symlink() and not output_path.exists() and
            output_path.parent.is_dir() and output_path.parent.resolve() == output_path.parent and
            output_path.suffix == '.json' and
            (output_path.is_relative_to(Path('/workspace')) or output_path.is_relative_to(Path('/tmp'))) and
            not output_path.is_relative_to(ROOT), 'Fresh exclusive preservation output outside repo')
    producer = regular(Path(__file__).resolve())
    producer_raw = digest(producer)
    modules_before = project_modules()
    frozen, baseline, pins = preserved_inputs()
    declared = dict(frozen)
    for name, expected in additional_source_sha256.items():
        require(name not in frozen, 'Authorized addition is genuinely new')
        pin(pins, ROOT / name, expected)
        declared[name] = expected
    exact(source_map(), declared, 'Actual887 equals all883 unchanged plus exactly frozen four')
    pin(pins, producer, producer_raw)
    require(str(output_path) not in pins, 'Fresh output disjoint from every preserved dependency')
    # Recheck all actual raw bytes and strict closures; use no prior main.
    after_frozen, after_baseline, after_pins = preserved_inputs()
    exact(after_frozen, frozen, 'Saved883 source authority unchanged')
    exact(after_baseline, baseline, 'Saved pre-new-source readback unchanged')
    for name, expected in pins.items():
        if name in after_pins:
            exact(after_pins[name], expected, 'Preserved raw dependency AFTER ' + name)
        else:
            exact(digest(name), expected, 'Declared new source and executing helper AFTER ' + name)
    exact(source_map(), declared, 'Complete actual887 source map unchanged AFTER')
    modules_after = project_modules()
    exact(set(modules_after), set(modules_before), 'Real caller project module names retained')
    require(all(modules_after[name] is module for name, module in modules_before.items()),
            'Real caller project module identities retained')
    result = dict(
        format='native-archive-material-carrier-prior883-preservation-1',
        status='PASS_IMMUTABLE883_WITH_EXACT_FOUR_DECLARED_SOURCE_ADDITIONS',
        current_python_source_count=len(declared), prior_python_source_count=883,
        prior_python_source_map_canonical_sha256=FROZEN_MAP_CANONICAL,
        current_python_source_map_canonical_sha256=value_digest(declared),
        additional_python_source_count=4, additional_python_source_sha256=additional_source_sha256,
        all883_python_sources_unchanged=True, complete_actual_source_map_checked_before_after=True,
        prior_883_metadata_is_authenticated_saved_readback_only=True,
        fresh_pre_new_source_883_readback_path=str(STAGE / 'baseline_883_readback.json'),
        fresh_pre_new_source_883_readback_raw_sha256=FRESH_REPORT_RAW,
        fresh_pre_new_source_883_readback_selfseal=FRESH_REPORT_SEAL,
        fresh_pre_new_source_883_readback_seconds=baseline['seconds'],
        preserved_prior_raw_dependency_pin_count=341, recorded_raw_dependency_pin_count=len(pins),
        dependency_raw_sha256=pins, preserved_current883_manifest_file_count=56,
        preserved_current883_manifest_raw_sha256=MANIFEST_RAW, preserved_current883_manifest_selfseal=MANIFEST_SEAL,
        preserved_current883_final_readback_archive_file_count=6,
        preserved_current883_final_readback_archive_raw_sha256=FINAL_ARCHIVE_RAW,
        preserved_inherited879_manifest_file_count=83,
        preserved_inherited879_final_readback_archive_file_count=6,
        authoritative_ancestry_release_closure_count=11, preserved_current883_release_closed_file_count=5,
        temporary_dependency_copy_association_count=15,
        temporary_dependency_exact_closed_copy_associations=baseline['temporary_dependency_exact_closed_copy_associations'],
        current_tmp_inputs_restored=False, project_modules_imported_reloaded_or_replaced=False,
        caller_existing_project_module_count=len(modules_before),
        prior_reader_or_auditor_main_reexecuted=False,
        math_or_solver_or_test_or_quadrature_reexecuted=False,
        executing_preservation_source_raw_sha256=producer_raw, seconds=time.perf_counter() - started,
    )
    result['preservation_sha256'] = value_digest(result)
    with output_path.open('xb') as stream:
        stream.write(encoded(result) + b'\n')
    return result
