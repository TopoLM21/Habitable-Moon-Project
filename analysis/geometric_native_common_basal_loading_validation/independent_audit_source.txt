BETA=F.from_float(1.2e14)

AMPLITUDE_GRADIENT=(F.from_float(.17),F(0),F(0))

def encoded(x):
    return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()

def digest(x):
    return hashlib.sha256(encoded(x)).hexdigest()

def raw_sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as stream:
        while block:=stream.read(1<<20):h.update(block)
    return h.hexdigest()

def require(ok,label):
    if not ok:raise AssertionError(label)

def rat(x):
    require(type(x) is not bool,'Boolean is not a scalar')
    if type(x) is list:
        require(len(x)==2 and all(type(v) is int for v in x) and x[1]>0,'Exact rational pair')
        return F(*x)
    return F.from_float(x) if type(x) is float else F(x)

def number(x):
    if type(x) is dict and set(x)=={'lo','hi'}:
        x=(rat(x['lo'])+rat(x['hi']))/2
    else:x=rat(x)
    return mp.mpf(x.numerator)/x.denominator

def vector(x):return tuple(number(v) for v in x)

def bounds(x):
    if type(x) is dict and set(x)=={'lo','hi'}:return number(x['lo']),number(x['hi'])
    v=number(x);return v,v

def compare(value,raw,label,counts):
    lo,hi=bounds(raw)
    margin=TOL*max(mp.mpf(1),abs(value),abs(lo),abs(hi))
    require(lo-margin<=value<=hi+margin,label+' lies outside registered enclosure')
    counts['scalar_enclosure_checks']+=1

def identity(value,label,counts,scale=1):
    require(abs(value)<=TOL*max(mp.mpf(1),abs(scale)),label)
    counts['analytic_identity_checks']+=1

def dot(a,b):return sum((x*y for x,y in zip(a,b)),mp.mpf(0))

def cross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])

def add(a,b):return tuple(x+y for x,y in zip(a,b))

def scaled(a,s):return tuple(s*x for x in a)


def source_at(y,radius):
    """Independent scalar differentiation of fixed-global angular source."""
    norm=mp.sqrt(dot(y,y));require(norm>0,'Positive actual spatial radius')
    q=scaled(y,1/norm)
    beta=number(BETA);omega=vector(OMEGA);gradient=vector(AMPLITUDE_GRADIENT)
    a=number(AMPLITUDE_CONSTANT)+dot(gradient,q)
    tau=scaled(cross(omega,q),beta*radius*a)
    columns=[]
    for k in range(3):
        direction=tuple(mp.mpf(i==k) for i in range(3))
        dq=scaled(add(direction,scaled(q,-q[k])),1/norm)
        dtau=scaled(add(scaled(cross(omega,q),dot(gradient,dq)),scaled(cross(omega,dq),a)),beta*radius)
        columns.append(dtau)
    dtau=tuple(tuple(columns[j][i] for j in range(3)) for i in range(3))
    return dict(q=q,norm=norm,amplitude=a,tau=tau,dtau=dtau)


def material_record(y,J,H,weight,radius,rates):
    """Compute one sparse material record, preserving all local coordinate jets."""
    n=len(J);require(len(H)==n and all(len(row)==n for row in H),'Complete local kinematic source')
    source=source_at(y,radius);beta=number(BETA)
    u=tuple(sum((J[i][a]*rates[i] for i in range(n)),mp.mpf(0)) for a in range(3))
    transmitted=add(source['tau'],scaled(u,-beta))
    L=tuple(weight*dot(J[i],source['tau']) for i in range(n))
    D=tuple(tuple(weight*beta*dot(J[i],J[j]) for j in range(n)) for i in range(n))
    Q=tuple(weight*dot(J[i],transmitted) for i in range(n))
    changes=[]
    for k in range(n):
        du=tuple(sum((H[j][k][a]*rates[j] for j in range(n)),mp.mpf(0)) for a in range(3))
        dtau=tuple(dot(source['dtau'][a],J[k]) for a in range(3))
        changes.append(add(dtau,scaled(du,-beta)))
    dQ=tuple(tuple(weight*(dot(H[i][k],transmitted)+dot(J[i],changes[k])) for k in range(n)) for i in range(n))
    return dict(source=source,u=u,transmitted=transmitted,L=L,D=D,Q=Q,dQ=dQ,
        driving_power=weight*dot(source['tau'],u),dissipation=weight*beta*dot(u,u),
        net_power=weight*dot(transmitted,u))

def local_taylor(y,J,H,delta):
    """Local diagnostic chart retaining the actual complete first/second jets."""
    n=len(J)
    yy=tuple(y[a]+sum((J[i][a]*delta[i] for i in range(n)),mp.mpf(0))
        +sum((H[i][j][a]*delta[i]*delta[j]/2 for i in range(n) for j in range(n)),mp.mpf(0)) for a in range(3))
    JJ=tuple(tuple(J[i][a]+sum((H[i][j][a]*delta[j] for j in range(n)),mp.mpf(0)) for a in range(3)) for i in range(n))
    return yy,JJ


def local_Q(y,J,H,weight,radius,rates,i,k,t):
    # Only coordinate k is displaced, so no algebraic dense chart evaluation
    # supplies the differentiated formula being checked.
    yy=tuple(y[a]+J[k][a]*t+H[k][k][a]*t*t/2 for a in range(3))
    JJ=tuple(tuple(J[j][a]+H[j][k][a]*t for a in range(3)) for j in range(len(J)))
    q=scaled(yy,1/mp.sqrt(dot(yy,yy)))
    tau=scaled(cross(vector(OMEGA),q),number(BETA)*radius*(number(AMPLITUDE_CONSTANT)+dot(vector(AMPLITUDE_GRADIENT),q)))
    u=tuple(sum((JJ[j][a]*rates[j] for j in range(len(J))),mp.mpf(0)) for a in range(3))
    return weight*dot(JJ[i],add(tau,scaled(u,-number(BETA))))


def exact_contractions(J,rates,weight,traction):
    """Rational Gram/virtual-work controls on authenticated representatives."""
    n=len(J);r=tuple(rat(x) for x in rates)
    def center(v):return (rat(v['lo'])+rat(v['hi']))/2 if type(v) is dict else rat(v)
    j=tuple(tuple(center(v) for v in row) for row in J)
    w=center(weight);tau=tuple(center(v) for v in traction)
    require(w>0,'Strictly positive registered reference weight')
    u=tuple(sum((j[i][a]*r[i] for i in range(n)),F(0)) for a in range(3))
    diss=w*BETA*sum((x*x for x in u),F(0))
    contracted=sum((r[i]*r[k]*w*BETA*sum((j[i][a]*j[k][a] for a in range(3)),F(0)) for i in range(n) for k in range(n)),F(0))
    require(diss==contracted and diss>=0,'Exact101 drag contraction equals nonnegative spatial dissipation')
    driving=w*sum((tau[a]*u[a] for a in range(3)),F(0))
    generalized=sum((r[i]*w*sum((j[i][a]*tau[a] for a in range(3)),F(0)) for i in range(n)),F(0))
    require(driving==generalized,'Exact source virtual work equals full-coordinate contraction')
    slip=tuple(tau[a]-BETA*u[a] for a in range(3))
    net=w*sum((slip[a]*u[a] for a in range(3)),F(0))
    contracted_net=sum((r[i]*w*sum((j[i][a]*slip[a] for a in range(3)),F(0)) for i in range(n)),F(0))
    require(net==contracted_net and net==driving-diss,'Exact net virtual work and source-minus-dissipation identity')
    return driving,diss,net


class QI:
    """Independent exact rational interval algebra, no repository dependency."""
    __slots__=('lo','hi')
    def __init__(self,lo=0,hi=None):
        self.lo=F(lo);self.hi=self.lo if hi is None else F(hi)
        require(self.lo<=self.hi,'Ordered rational enclosure')
    def __add__(self,other):
        other=qi(other);return QI(self.lo+other.lo,self.hi+other.hi)
    __radd__=__add__
    def __neg__(self):return QI(-self.hi,-self.lo)
    def __sub__(self,other):return self+-qi(other)
    def __rsub__(self,other):return qi(other)+-self
    def __mul__(self,other):
        other=qi(other);v=(self.lo*other.lo,self.lo*other.hi,self.hi*other.lo,self.hi*other.hi)
        return QI(min(v),max(v))
    __rmul__=__mul__
    def __truediv__(self,other):
        other=qi(other);require(not other.lo<=0<=other.hi,'Nonzero interval divisor')
        return self*QI(1/other.hi,1/other.lo)
    def __rtruediv__(self,other):return qi(other)/self


def qi(v):
    if type(v) is QI:return v
    if type(v) is dict and set(v)=={'lo','hi'}:return QI(rat(v['lo']),rat(v['hi']))
    return QI(rat(v))

def qsquare(v):
    v=qi(v);return QI(0 if v.lo<=0<=v.hi else min(v.lo*v.lo,v.hi*v.hi),max(v.lo*v.lo,v.hi*v.hi))

def qlog2(v):
    require(v>0,'Positive rational scale')
    a,b=v.numerator,v.denominator;e=a.bit_length()-b.bit_length()
    return e-int(a<(b<<e) if e>=0 else (a<<(-e))<b)

def qpower2(e):return F(1<<e) if e>=0 else F(1,1<<(-e))

def qround(v,precision):
    # Carrier compression has sixteen additional decimal guard digits.
    v=qi(v);maximum=max(abs(v.lo),abs(v.hi))
    if maximum==0:return QI(0)
    bits=(3322*(precision+16)+999)//1000+8
    scale=qpower2(bits-qlog2(maximum));l=v.lo*scale;h=v.hi*scale
    return QI(F(l.numerator//l.denominator)/scale,F(-((-h.numerator)//h.denominator))/scale)

def qsqrt(v,precision):
    v=qi(v);require(v.lo>=0,'Nonnegative radial square')
    def point(x):
        if x==0:return QI(0)
        a,b=isqrt(x.numerator),isqrt(x.denominator)
        if a*a==x.numerator and b*b==x.denominator:return QI(F(a,b))
        bits=(3322*precision+999)//1000+8
        scale=qpower2(bits-qlog2(x)//2);s=x*scale*scale;r=isqrt(s.numerator//s.denominator)
        z=QI(F(r)/scale,F(r+1)/scale)
        require(z.lo*z.lo<=x<=z.hi*z.hi,'Exact integer square-root enclosure')
        return z
    l,h=point(v.lo),point(v.hi);return QI(l.lo,h.hi)

def qdot(a,b):return sum((x*y for x,y in zip(a,b)),QI(0))

def qcross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])

def qintersection(a,b):
    require(a.lo<=b.hi and b.lo<=a.hi,'Independent interval overlap')
    return QI(max(a.lo,b.lo),min(a.hi,b.hi))

def qequal(actual,raw,label,counts):
    raw=qi(raw);require(actual.lo==raw.lo and actual.hi==raw.hi,label+' exact rational replay mismatch')
    counts['exact_rational_operator_enclosure_replays']+=1

def qsource(position,radius,precision):
    x=tuple(qi(v) for v in position);r=qsqrt(sum((qsquare(v) for v in x),QI(0)),precision+16)
    require(r.lo>0,'Authentic material point separated from origin')
    q=tuple(qround(v/r,precision) for v in x);omega=tuple(QI(v) for v in OMEGA);g=tuple(QI(v) for v in AMPLITUDE_GRADIENT)
    a=QI(AMPLITUDE_CONSTANT)+qdot(g,q);c=qcross(omega,q);calibration=BETA*radius
    tau=tuple(qround(calibration*a*v,precision) for v in c)
    W=tuple(tuple(qcross(omega,tuple(QI(int(l==j)) for l in range(3)))[i] for j in range(3)) for i in range(3))
    P=tuple(tuple(QI(int(i==j))-q[i]*q[j] for j in range(3)) for i in range(3))
    derivative=tuple(tuple(qround(calibration/r*sum(((c[i]*g[l]+a*W[i][l])*P[l][j] for l in range(3)),QI(0)),precision) for j in range(3)) for i in range(3))
    return tau,derivative,dict(current_radius_bounds_m=r,current_unit_direction_bounds=q,amplitude_bounds=a)

def qreplay(records,response,radius,counts):
    """Full exactFraction enclosure replay of each authentic trace contribution."""
    n=101;p=response['precision'];rates=tuple(rat(v) for v in response['coordinate_rates'])
    L=[QI(0) for _ in range(n)];Q=[QI(0) for _ in range(n)]
    D=[[QI(0) for _ in range(n)] for _ in range(n)];K=[[QI(0) for _ in range(n)] for _ in range(n)]
    ps,pd,pn=QI(0),QI(0),QI(0)
    receipts=response['per_trace_receipts_json'];require(len(receipts)==len(records),'Exactly one operator receipt per positive source row')
    for record,receipt_raw in zip(records,receipts):
        receipt=json.loads(receipt_raw) if type(receipt_raw) is str else receipt_raw
        ids=record['global_coordinate_indices'];m=len(ids);w=qi(record['source_weight_bounds_m2'])
        J=tuple(tuple(qi(v) for v in row) for row in record['position_first_bounds'])
        H=tuple(tuple(tuple(qi(v) for v in row) for row in plane) for plane in record['position_second_bounds'])
        tau,dtau,src=qsource(record['position_bounds_m'],radius,p)
        u=tuple(qround(sum((J[a][j]*rates[g] for j,g in enumerate(ids)),QI(0)),p) for a in range(3))
        slip=tuple(qround(t-BETA*v,p) for t,v in zip(tau,u))
        cols=tuple(tuple(J[a][i] for a in range(3)) for i in range(m))
        changes=[]
        for k in range(m):
            du=tuple(qround(sum((H[a][j][k]*rates[g] for j,g in enumerate(ids)),QI(0)),p) for a in range(3))
            changes.append(tuple(qdot(row,cols[k])-BETA*d for row,d in zip(dtau,du)))
        for i,g in enumerate(ids):
            L[g]=qround(L[g]+qround(w*qdot(cols[i],tau),p),p)
            Q[g]=qround(Q[g]+qround(w*qdot(cols[i],slip),p),p)
            for j,h in enumerate(ids):
                if j>=i:
                    D[g][h]=qround(D[g][h]+qround(BETA*w*qdot(cols[i],cols[j]),p),p)
                    if h!=g:D[h][g]=D[g][h]
                hess=tuple(H[a][i][j] for a in range(3))
                K[g][h]=qround(K[g][h]+qround(w*(qdot(hess,slip)+qdot(cols[i],changes[j])),p),p)
        s=qround(w*qdot(tau,u),p);d=qround(BETA*w*sum((qsquare(v) for v in u),QI(0)),p);net=qround(w*qdot(slip,u),p)
        ps=qround(ps+s,p);pd=qround(pd+d,p);pn=qround(pn+net,p)
        for field,actual in (('prescribed_driving_traction_bounds_pa',tau),('actual_material_velocity_bounds_m_s',u),('net_basal_traction_bounds_pa',slip)):
            for a in range(3):qequal(actual[a],receipt[field][a],record.get('trace_id','trace')+'.'+field,counts)
        for a in range(3):
            qequal(src['current_unit_direction_bounds'][a],receipt['current_unit_direction_bounds'][a],'Current q',counts)
            for b in range(3):qequal(dtau[a][b],receipt['prescribed_spatial_derivative_bounds_pa_m'][a][b],'Nonsymmetric Cartesian source derivative',counts)
        for field,actual in (('source_power_bounds_w',s),('drag_dissipation_bounds_w',d),('net_power_bounds_w',net),('current_radius_bounds_m',src['current_radius_bounds_m']),('amplitude_bounds',src['amplitude_bounds'])):
            qequal(actual,receipt[field],field,counts)
    gs=qround(sum((L[i]*rates[i] for i in range(n)),QI(0)),p)
    gd=qround(sum((rates[i]*D[i][j]*rates[j] for i in range(n) for j in range(n)),QI(0)),p)
    gn=qround(sum((Q[i]*rates[i] for i in range(n)),QI(0)),p)
    finalsource=qintersection(ps,gs);finaldiss=qintersection(pd,gd);finalnet=qintersection(qintersection(pn,gn),finalsource-finaldiss)
    contracted=tuple(qround(L[i]-sum((D[i][j]*rates[j] for j in range(n)),QI(0)),p) for i in range(n))
    residual=tuple(qintersection(Q[i],contracted[i]) for i in range(n))
    for field,values in (('prescribed_generalized_force_bounds',L),('residual_bounds',residual)):
        for i in range(n):qequal(values[i],response[field][i],field+'.'+str(i),counts)
    for field,values in (('drag_bounds',D),('state_tangent_bounds',K),('velocity_tangent_bounds',[[ -x for x in row ] for row in D])):
        for i in range(n):
            for j in range(n):qequal(values[i][j],response[field][i][j],field+'.'+str(i)+','+str(j),counts)
    for field,v in (('source_power_bounds_w',finalsource),('drag_dissipation_bounds_w',finaldiss),('net_power_bounds_w',finalnet),('independently_projected_source_power_bounds_w',ps),('independently_projected_drag_dissipation_bounds_w',pd),('independently_projected_net_power_bounds_w',pn),('generalized_source_power_bounds_w',gs),('generalized_drag_dissipation_bounds_w',gd),('generalized_net_power_bounds_w',gn)):
        qequal(v,response[field],field,counts)
    require(finaldiss.lo>=0,'Rigorous nonnegative complete declared drag dissipation')
    return dict(source_power=finalsource,dissipation=finaldiss,net_power=finalnet)


def decoded_record(record):
    ids=tuple(record['global_coordinate_indices']);m=len(ids)
    require(len(set(ids))==m and all(type(i) is int and 0<=i<101 for i in ids),'Complete distinct material coordinate support')
    Jraw=record['position_first_bounds'];Hraw=record['position_second_bounds']
    require(len(Jraw)==3 and all(len(row)==m for row in Jraw),'Position first jet3xN')
    require(len(Hraw)==3 and all(len(plane)==m and all(len(row)==m for row in plane) for plane in Hraw),'Position second jet3xNxN')
    require(all(Hraw[a][i][j]==Hraw[a][j][i] for a in range(3) for i in range(m) for j in range(m)),'Authentic position curvature symmetric')
    J=tuple(tuple(number(Jraw[a][i]) for a in range(3)) for i in range(m))
    H=tuple(tuple(tuple(number(Hraw[a][i][j]) for a in range(3)) for j in range(m)) for i in range(m))
    return ids,vector(record['position_bounds_m']),J,H


def mpaudit(records,response,radius,counts,*,derivatives=True):
    n=101;rates=tuple(rat(v) for v in response['coordinate_rates'])
    L=[mp.mpf(0) for _ in range(n)];Q=[mp.mpf(0) for _ in range(n)]
    D=[[mp.mpf(0) for _ in range(n)] for _ in range(n)];K=[[mp.mpf(0) for _ in range(n)] for _ in range(n)]
    driving=diss=net=mp.mpf(0);mp_radius=number(radius);selected=[]
    receipts=response['per_trace_receipts_json']
    for k,(record,receipt_raw) in enumerate(zip(records,receipts)):
        receipt=json.loads(receipt_raw) if type(receipt_raw) is str else receipt_raw
        ids,y,J,H=decoded_record(record);weight=number(record['source_weight_bounds_m2'])
        local_rates=tuple(number(rates[g]) for g in ids)
        math=material_record(y,J,H,weight,mp_radius,local_rates)
        for i,g in enumerate(ids):
            L[g]+=math['L'][i];Q[g]+=math['Q'][i]
            for j,h in enumerate(ids):D[g][h]+=math['D'][i][j];K[g][h]+=math['dQ'][i][j]
        driving+=math['driving_power'];diss+=math['dissipation'];net+=math['net_power']
        for a in range(3):
            compare(math['source']['q'][a],receipt['current_unit_direction_bounds'][a],'220digit actual q',counts)
            compare(math['source']['tau'][a],receipt['prescribed_driving_traction_bounds_pa'][a],'220digit actual fixedR driving',counts)
            compare(math['u'][a],receipt['actual_material_velocity_bounds_m_s'][a],'220digit actual spatial material velocity',counts)
            compare(math['transmitted'][a],receipt['net_basal_traction_bounds_pa'][a],'220digit independent full-vector slip',counts)
            for j in range(3):compare(math['source']['dtau'][a][j],receipt['prescribed_spatial_derivative_bounds_pa_m'][a][j],'220digit actual nonsymmetric source derivative',counts)
        require(math['dissipation']>=0,'Independent positive spatial dissipation')
        require(receipt['amplitude_multiplies_drag'] is False and receipt['spatial_derivative_assumed_symmetric'] is False,'Independent driving amplitude and full unsymmetric spatial derivative')
        rawcolumns=tuple(tuple(record['position_first_bounds'][a][i] for a in range(3)) for i in range(len(ids)))
        exact_contractions(rawcolumns,tuple(rates[g] for g in ids),record['source_weight_bounds_m2'],receipt['prescribed_driving_traction_bounds_pa'])
        counts['exact_rational_algebra_controls']+=4
        # Controls retain actual supplied geometry, source radius, all local
        # rates and the exact second jet. They differentiate the scalar law,
        # independently of its analytic derivative implementation.
        kind=record.get('material_kind',record.get('source_material_kind',''))
        chosen=[i for i,g in enumerate(ids) if g<6][:1]+[i for i,g in enumerate(ids) if 6<=g<92][:2]+[i for i,g in enumerate(ids) if g==100]
        if derivatives and (k==0 or ('hot' in str(kind).lower() and not any(v.get('hot') for v in selected))):
            for i in chosen:
                for j in chosen:
                    expected=math['dQ'][i][j]
                    got=mp.diff(lambda t:local_Q(y,J,H,weight,mp_radius,local_rates,i,j,t),mp.mpf(0))
                    identity(got-expected,'Independent scalar material basal dQ/dx',counts,max(abs(got),abs(expected)))
                    counts['independent_mpmath_derivative_controls']+=1
            selected.append(dict(trace_id=record.get('trace_id'),global_coordinate_controls=[ids[i] for i in chosen],hot='hot' in str(kind).lower()))
    for i in range(n):
        compare(L[i],response['prescribed_generalized_force_bounds'][i],'Full101 prescribed generalized load',counts)
        compare(Q[i],response['residual_bounds'][i],'Full101 direct residual',counts)
        for j in range(n):
            compare(D[i][j],response['drag_bounds'][i][j],'Full101x101 independent Gram drag',counts)
            compare(K[i][j],response['state_tangent_bounds'][i][j],'Full101x101 material prestress/spatial/drag tangent',counts)
            compare(-D[i][j],response['velocity_tangent_bounds'][i][j],'Full101x101 velocity tangent',counts)
    rr=tuple(number(x) for x in rates)
    driving_contracted=sum((rr[i]*L[i] for i in range(n)),mp.mpf(0))
    diss_contracted=sum((rr[i]*D[i][j]*rr[j] for i in range(n) for j in range(n)),mp.mpf(0))
    net_contracted=sum((rr[i]*Q[i] for i in range(n)),mp.mpf(0))

def rodrigues(v):
    v=mp.matrix(v);K=mp.matrix(((0,-v[2],v[1]),(v[2],0,-v[0]),(-v[1],v[0],0)))

    Q=mp.eye(3)+mp.sin(angle)/angle*K+(1-mp.cos(angle))/(angle*angle)*(K*K)
    return tuple(tuple(Q[i,j] for j in range(3)) for i in range(3))

def seam_axis(ray,numerator,denominator,delta):
    a,b=ray[numerator],abs(ray[denominator])
    if b==0:return mp.mpf('0.5') if a==0 else mp.mpf(a>0)
    angle=mp.atan(a/b)
    if angle<=-delta:return mp.mpf(0)
    if angle>=delta:return mp.mpf(1)
    t=(angle+delta)/(2*delta);return t**3*(10+t*(-15+6*t))
