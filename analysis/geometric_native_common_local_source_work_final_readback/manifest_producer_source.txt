    return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()

def value_digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()

def require(condition,label):
    if not condition:raise ValueError(label)

def exact(actual,expected,label):
    require(type(actual) is type(expected) and actual==expected,label)
    if type(actual) in (dict,list,tuple):require(encoded(actual)==encoded(expected),label+' complete literal encoding')

def pin_valid(pin):
    return type(pin) is str and len(pin)==64 and all(c in '0123456789abcdef' for c in pin)

def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda:stream.read(1<<20),b''):h.update(block)
    return h.hexdigest()

def read(path):
    return json.loads(Path(path).read_bytes())

def sealed(path,key):

def static(node,values):
    if isinstance(node,ast.Name):return values[node.id]
    if isinstance(node,ast.Dict):return {static(k,values):static(v,values) for k,v in zip(node.keys,node.values)}
    if isinstance(node,(ast.Tuple,ast.List,ast.Set)):
        return {ast.Tuple:tuple,ast.List:list,ast.Set:set}[type(node)](static(v,values) for v in node.elts)
    if isinstance(node,ast.BinOp) and isinstance(node.op,ast.Div):return static(node.left,values)/static(node.right,values)
    if isinstance(node,ast.Call) and isinstance(node.func,ast.Name):
        args=[static(x,values) for x in node.args]
        if node.func.id=='dict':return dict(*args,**{k.arg:static(k.value,values) for k in node.keywords})
        constructors={'Path':Path,'str':str,'frozenset':frozenset,'list':list,'tuple':tuple,'sorted':sorted}
        if node.func.id in constructors:return constructors[node.func.id](*args)
    return ast.literal_eval(node)

def constants(path):
    tree=ast.parse(Path(path).read_bytes());values={'ROOT':ROOT}
    for node in tree.body:
        if isinstance(node,ast.Assign):
            try:value=static(node.value,values)
            except (ValueError,KeyError,TypeError):continue
            for target in node.targets:
                if isinstance(target,ast.Name):values[target.id]=value
    return values,tree

          'all7 auditor schemas equal frozen production field declarations')
    # _FLAGS is a literal dict(...) declaration, inspected without importing it.
    flag_assignments = [node for node in core_tree.body
        if type(node).__name__ == 'Assign' and any(
            type(target).__name__ == 'Name' and target.id == '_FLAGS'
            for target in node.targets)]

    def rows(value, count, label):

    def fraction(value, label):
        rows(value, 2, label)
        require(all(type(item) is int for item in value) and value[1] > 0,
                label + ': literal rational numerator/positive denominator')
        a, b = abs(value[0]), value[1]
        while b:
            a, b = b, a % b
        exact(a, 1, label + ': canonical reduced rational')
        return value

    def interval(value, label):
        require(type(value) is dict, label + ': literal interval object')
        exact(set(value), {'lo', 'hi'}, label + ': interval endpoint schema')
        lo, hi = fraction(value['lo'], label + ':lo'), fraction(value['hi'], label + ':hi')
        require(lo[0] * hi[1] <= hi[0] * lo[1], label + ': ordered interval endpoints')
        return value

    def shape(value, dimensions, scalar, label):
        if not dimensions:
            return scalar(value, label)
        rows(value, dimensions[0], label)
        for index, item in enumerate(value):
            shape(item, dimensions[1:], scalar, label + ':' + str(index))
        return value

    def contains(outer, inner, label):
        interval(outer, label + ':packet')
        interval(inner, label + ':audit')

        exact(raw['binding_sha256'], value_digest({key: raw[key] for key in declared[name]
              if key != 'binding_sha256'}), 'entire typed DTO selfseal: ' + name)
        return raw

    packet_extra = {'format', 'prior_875_packet_raw_sha256', 'prior_875_packet_selfseal',
