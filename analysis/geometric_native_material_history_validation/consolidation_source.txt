def encoded(value):
    return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def read(path):
    return json.loads((ROOT/path).read_bytes())

def seal_valid(raw,key):
    assert hashlib.sha256(encoded({k:v for k,v in raw.items() if k!=key})).hexdigest()==raw[key],key

def write(name,value):
    with (OUT/name).open('x') as stream:
        json.dump(value,stream,indent=2,sort_keys=True,allow_nan=False)
        stream.write('\n')

def interval(raw):
    return Fraction(*raw['lo']),Fraction(*raw['hi'])

def baseline(path):
    sources=read(path)
    changed=[p for p,v in sources.items() if digest(ROOT/p)!=v]
    return dict(path=path,sha256=digest(ROOT/path),count=len(sources),changed_paths=changed)
