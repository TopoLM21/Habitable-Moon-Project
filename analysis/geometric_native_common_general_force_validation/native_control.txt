def encoded(value):
    return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()

def digest(value):
    return hashlib.sha256(encoded(value)).hexdigest()

def source(path,pin):
    raw=path.read_bytes()
    if hashlib.sha256(raw).hexdigest()!=pin:raise ValueError('A whole immutable source authority changed: '+str(path))
    return json.loads(raw)

def query_states(ctx):
    hot=tuple(F(*v) for v in ctx['source903']['evaluation_queries'][1]['response']['coordinate_increments'])
    cold=tuple(v/4 for v in ctx['cold_carrier'].bound_coordinate_increments)
    owner=tuple(F((-1 if i%2 else 1)*(i+1),10000) for i in range(6))
    internal=tuple(F((-1 if i%2 else 1)*(i+1),1000000) for i in range(8))
    joint=owner+hot+cold+internal+(F(1,2),)
    rates=tuple(F((-1 if i%2 else 1)*(i+1),10**16) for i in range(6))
    rates+=tuple(F((-1 if i%2 else 1)*(i+1),10**15) for i in range(94))+(F(2,10**12),)
    return [('zero_source_zero_rates',(F(0),)*101,(F(0),)*101),
            ('finite_owner_joint_body_preparation_rates',joint,rates),
            ('same_joint_state_opposite_rates',joint,tuple(-v for v in rates))]

def joint_point_controls(ctx):
    from tectonics import geometric_native_common_coupled_cold_energy as api
    system=ctx['cold_energy_system'];port=api.native_common_coupled_cold_energy_context(system)
    archive=port['archive'];materials=ctx['continuum_system'].banks
    donor=next(b for b in materials if b.owner==archive.donor_owner)
    receiver=next(b for b in materials if b.owner==archive.receiver_owner)
    delta=ctx['cold_carrier'].whole_domain_certificate.seam_half_width_radians
    c=tuple(v/4 for v in ctx['cold_carrier'].bound_coordinate_increments)
    specs=[('donor_interior_upper_zero',donor,'interior',F(1,3),'upper',(F(0),)*56,F(0)),
           ('receiver_start_seam_upper_joint',receiver,'start_seam',delta/2,'upper',c,F(1,2)),
           ('donor_end_seam_lower_joint',donor,'end_seam',delta/2,'lower',c,F(1,2)),
           ('receiver_interior_lower_full_preparation',receiver,'interior',F(1,3),'lower',c,F(1))]
    controls=[]
    for label,material,tag,angle,layer,coordinates,lam in specs:
        bank=next(b for b in ctx['cold_carrier'].banks if b.original_parent_id==material.original_parent_id)
        blo=2*archive.whole_domain_certificate.gamma if material.owner==archive.donor_owner else F(0)
        beta=(blo+bank.beta_cut_radians)/2
        depth=material.elastic_layer_areal_depth_m/2 if layer=='upper' else (
            material.elastic_layer_areal_depth_m+material.areal_material_thickness_m)/2
        point=api.evaluate_native_common_coupled_cold_point(system,material.original_parent_id,tag,angle,beta,depth,
            coordinates,lam,precision=96,layer=layer)
        controls.append(dict(label=label,response=api.native_common_coupled_cold_point_to_dict(point)))
    return controls
