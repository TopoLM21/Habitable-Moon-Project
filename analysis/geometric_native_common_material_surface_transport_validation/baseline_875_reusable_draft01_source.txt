def enc(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()


def vd(value):
    return hashlib.sha256(enc(value)).hexdigest()


def exact(actual, expected, label):
    assert type(actual) is type(expected) and actual == expected, label
    if type(actual) in (dict, list, tuple):
        assert enc(actual) == enc(expected), label + ' recursive literal kinds'


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def sealed(path, key):
    value = json.loads(Path(path).read_bytes())
    body = dict(value)
    exact(body.pop(key), vd(body), 'selfseal ' + str(path))
    return value


def source_map():
    return {str(path.relative_to(ROOT)): digest(path)
        for directory in ('tectonics', 'tests', 'scripts', 'analysis')
        for path in sorted((ROOT / directory).rglob('*.py'))}


def project_modules():
    result = {}
    for name, module in sys.modules.items():
        location = getattr(module, '__file__', None)
        path = Path(location).resolve() if isinstance(location, str) else None
        project_file = path is not None and path.suffix == '.py' and path.is_relative_to(ROOT)
        project_namespace = name == 'tectonics' or name.startswith('tectonics.') or name == 'tests' or name.startswith('tests.')
        if project_file or project_namespace:
            result[name] = module
    return result


def normalized_paths(paths):
    assert not isinstance(paths, (str, bytes)), 'Declare exactly four authorized additive path strings'
    names = tuple(paths)
    assert len(names) == len(set(names)) == 4, 'Four unique explicitly declared additions required'
    for name in names:
        assert type(name) is str and name
        path = PurePosixPath(name)
        assert not path.is_absolute() and str(path) == name and '..' not in path.parts, name
        assert len(path.parts) >= 2 and path.parts[0] in ('tectonics', 'tests', 'scripts', 'analysis') and path.suffix == '.py', name
    declared = frozenset(names)
    exact(declared, AUTHORIZED_ADDITIONAL_SOURCE_PATHS, 'Exact authorized material-surface source whitelist')
    return declared
