def enc(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()


def vd(value):
    return hashlib.sha256(enc(value)).hexdigest()


def exact(actual, expected, label):
    assert type(actual) is type(expected) and actual == expected, label
    if type(actual) in (dict, list, tuple):
        assert enc(actual) == enc(expected), label + ' recursive literal kinds'


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def sealed(path, key):
    value = read(path)
    assert type(value) is dict and key in value, str(path)
    body = dict(value)
    exact(body.pop(key), vd(body), 'selfseal ' + str(path))
    return value


def source_map():
    return {str(p.relative_to(ROOT)): digest(p)
        for directory in ('tectonics', 'tests', 'scripts', 'analysis')
        for p in sorted((ROOT / directory).rglob('*.py'))}


def add_pin(pins, path, pin):
    path = Path(path).resolve()
    assert path.is_relative_to(ROOT) or path.is_relative_to(Path('/tmp')), str(path)
    assert type(pin) is str and len(pin) == 64, str(path)
    name = str(path)
    if name in pins:
        exact(pins[name], pin, 'overlapping raw pin ' + name)
    pins[name] = pin


def files_from_manifest(manifest, pins):
    for name, row in manifest['files'].items():
        path = (ROOT / name).resolve()
        assert path.is_relative_to(ROOT) and str(path.relative_to(ROOT)) == name, name
        assert type(row['bytes']) is int and row['bytes'] >= 0, name
        exact(path.stat().st_size, row['bytes'], 'literal artifact bytes ' + name)
        add_pin(pins, path, row['sha256'])

def literal_constants(path):
    tree = ast.parse(Path(path).read_bytes(), filename=str(path))
    values = {}
    for node in tree.body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            try:
                values[node.targets[0].id] = ast.literal_eval(node.value)
            except (TypeError, ValueError):
                if isinstance(node.value, ast.Call) and isinstance(node.value.func, ast.Name) and node.value.func.id == 'dict':
                    try:
                        values[node.targets[0].id] = {k.arg: ast.literal_eval(k.value) for k in node.value.keywords if k.arg is not None}
                    except (TypeError, ValueError):
                        pass
    return values, tree


def typed_seal(value, name, schemas, constants, fmt):
    fields = set(schemas[name])
    exact(set(value), fields | {'format', 'scope', 'model'} | set(constants['_FLAGS']), 'typed complete keys ' + name)
    exact(value['format'], fmt, 'typed format ' + name)
    exact(value['scope'], SCOPE, 'typed scope ' + name)
    exact(value['model'], constants['MODEL'], 'typed model ' + name)
    for key, flag in constants['_FLAGS'].items():
        exact(value[key], flag, 'typed scope flag ' + key)
    body = {key: value[key] for key in schemas[name] if key != 'binding_sha256'}
    exact(value['binding_sha256'], vd(body), 'complete typed binding seal ' + name)
