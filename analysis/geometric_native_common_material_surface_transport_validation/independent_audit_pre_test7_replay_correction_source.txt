def enc(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def vd(v):return hashlib.sha256(enc(v)).hexdigest()
def digest(p):
 h=hashlib.sha256()
 with Path(p).open('rb') as f:
  for b in iter(lambda:f.read(1<<20),b''):h.update(b)
 return h.hexdigest()
def read(p):return json.loads(Path(p).read_bytes())
def payload(v):
 if type(v) is F:return [v.numerator,v.denominator]
 if hasattr(v,'lo') and hasattr(v,'hi'):return dict(lo=payload(v.lo),hi=payload(v.hi))
 if type(v) in (tuple,list):return [payload(x) for x in v]
 if type(v) is dict:return {k:payload(x) for k,x in v.items()}
 return v

def exact(a,b,label):
 assert type(a) is type(b) and a==b,label

def frac(v):
 assert type(v) is list and len(v)==2 and all(type(x) is int for x in v) and v[1]>0
 q=F(*v);exact(v,[q.numerator,q.denominator],'canonical literal Fraction');return q

def literal(v):

def interval(v,c):

def rounded(v,c,precision=256):

def square(v,c):

def sqrt_interval(v,c,precision=300):
 assert v.lo>=0
 scale=10**precision
 def floorroot(q):return math.isqrt((q*scale*scale).numerator//(q*scale*scale).denominator)

 if F(upper,scale)**2<v.hi:upper+=1
 assert F(lower,scale)**2<=v.lo and F(upper,scale)**2>=v.hi
 return c['I'](F(lower,scale),F(upper,scale))


def dot(a,b):return sum((x*y for x,y in zip(a,b)),F(0))
def scale(v,a):return tuple(x*a for x in v)
def add(a,b):return tuple(x+y for x,y in zip(a,b))
def cross(a,b):return (a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0])

def transpose(M):return tuple(tuple(M[j][i] for j in range(len(M))) for i in range(len(M[0])))
def matmul(A,B):return tuple(tuple(dot(row,col) for col in transpose(B)) for row in A)

@dataclass(frozen=True)
class Jet2:
 """Independent full value, gradient and symmetric Hessian body-coordinate jet."""
 value: object
 gradient: tuple
 hessian: tuple


def jconstant(v,c,n=7):
 I=c['I'];return Jet2(v if hasattr(v,'lo') else I(v),(I(0),)*n,tuple((I(0),)*n for _ in range(n)))

def jlinear(v,derivatives,c):
 I=c['I'];n=len(derivatives);return Jet2(v,tuple(derivatives),tuple((I(0),)*n for _ in range(n)))

def jadd(a,b,c):
 n=len(a.gradient);assert n==len(b.gradient)
 return Jet2(rounded(a.value+b.value,c),tuple(rounded(a.gradient[i]+b.gradient[i],c) for i in range(n)),tuple(tuple(rounded(a.hessian[i][j]+b.hessian[i][j],c) for j in range(n)) for i in range(n)))

def jneg(a,c):
 return Jet2(-a.value,tuple(-x for x in a.gradient),tuple(tuple(-x for x in row) for row in a.hessian))

def jmul(a,b,c):
 n=len(a.gradient);assert n==len(b.gradient)
 return Jet2(rounded(a.value*b.value,c),tuple(rounded(a.gradient[i]*b.value+a.value*b.gradient[i],c) for i in range(n)),tuple(tuple(rounded(a.hessian[i][j]*b.value+a.gradient[i]*b.gradient[j]+a.gradient[j]*b.gradient[i]+a.value*b.hessian[i][j],c) for j in range(n)) for i in range(n)))

def jscale(a,v,c):return jmul(a,jconstant(v,c,len(a.gradient)),c)

def jinverse(a,c):
 assert not a.value.lo<=0<=a.value.hi;n=len(a.gradient);inv=1/a.value
 return Jet2(rounded(inv,c),tuple(rounded(-a.gradient[i]*inv*inv,c) for i in range(n)),tuple(tuple(rounded(2*a.gradient[i]*a.gradient[j]*inv*inv*inv-a.hessian[i][j]*inv*inv,c) for j in range(n)) for i in range(n)))

def jsqrt(a,c):
 assert a.value.lo>0;n=len(a.gradient);v=sqrt_interval(a.value,c)
 return Jet2(rounded(v,c),tuple(rounded(a.gradient[i]/(2*v),c) for i in range(n)),tuple(tuple(rounded(a.hessian[i][j]/(2*v)-a.gradient[i]*a.gradient[j]/(4*v*v*v),c) for j in range(n)) for i in range(n)))

def jdot(a,b,c):
 assert len(a)==len(b);result=jconstant(0,c,len(a[0].gradient))
 for x,y in zip(a,b):result=jadd(result,jmul(x,y,c),c)
 return result


def jnorm_squared(vector,c):
 raw=jdot(vector,vector,c)
 value=rounded(sum((square(x.value,c) for x in vector),c['I'](0)),c)
 return Jet2(value,raw.gradient,raw.hessian)


def cofactor_jet(Fmatrix,c):
 """Signed 2x2 minors: cofactor(F)=det(F) F^{-T}, without matrix inverse."""
 rows=[]
 for i in range(3):
  row=[];rr=[r for r in range(3) if r!=i]
  for j in range(3):
   cc=[k for k in range(3) if k!=j]
   v=jadd(jmul(Fmatrix[rr[0]][cc[0]],Fmatrix[rr[1]][cc[1]],c),jneg(jmul(Fmatrix[rr[0]][cc[1]],Fmatrix[rr[1]][cc[0]],c),c),c)
   row.append(v if (i+j)%2==0 else jneg(v,c))
  rows.append(tuple(row))
 return tuple(rows)


def nanson_jet(gradient_matrix,mode_gradient_matrices,reference_normal,c):
 """Finite F is already I+du/dX; seven-mode F''=0, cofactor'' nonzero."""
 assert len(mode_gradient_matrices)==7
 Fmatrix=tuple(tuple(jlinear(gradient_matrix[i][j],tuple(M[i][j] for M in mode_gradient_matrices),c) for j in range(3)) for i in range(3));C=cofactor_jet(Fmatrix,c)
 determinant=jdot(Fmatrix[0],C[0],c);assert determinant.value.lo>0,'LOCAL finite incremental deformation preserves orientation'
 normal=tuple(jconstant(x,c) for x in reference_normal);area=tuple(jdot(row,normal,c) for row in C)
 norm2=jnorm_squared(area,c)
 # At exact identity F, the authentic reference unit normal has norm exactly1.
 # Preserve that correlated source identity rather than widen its known value.
 if all(gradient_matrix[i][j]==c['I'](int(i==j)) for i in range(3) for j in range(3)):
  norm2=Jet2(c['I'](1),norm2.gradient,norm2.hessian)
 js=jsqrt(norm2,c);unit=tuple(jmul(v,jinverse(js,c),c) for v in area)
 return dict(F=Fmatrix,cofactor=C,determinant=determinant,oriented_area=area,surface_ratio=js,current_normal=unit)


def convected_tangent_jet(Fmatrix,reference_tangent,c):

def trig_point(x,c):
 """Outward rational Taylor sums; explicit decreasing-tail bound, no floats."""
 I=c['I'];assert abs(x)<4
 if not x:return I(0),I(1)
 results=[]
 for sine in (True,False):
  term=I(x if sine else 1);partial=term;k=0
  while True:
   a=2*k+(2 if sine else 1);nextterm=rounded(-term*x*x/(a*(a+1)),c,320)
   tail=max(abs(nextterm.lo),abs(nextterm.hi));ratio=abs(x*x)/F((a+2)*(a+3))
   if ratio<1 and tail<F(1,10**310):
    results.append(rounded(partial+I(-tail/(1-ratio),tail/(1-ratio)),c,300));break
   partial=rounded(partial+nextterm,c,320);term=nextterm;k+=1
 return tuple(results)


def trig_range(a,c):
 """Endpoint monotonicity plus exact extrema, for source angles in [0,pi]."""
 I=c['I'];pi=c['geometry'].interval_pi(320);assert 0<=a.lo<=a.hi<=pi.hi
 sl,cl=trig_point(a.lo,c);sh,ch=trig_point(a.hi,c)
 sine=I(min(sl.lo,sh.lo),max(sl.hi,sh.hi));cosine=I(min(cl.lo,ch.lo),max(cl.hi,ch.hi))
 if a.lo<=pi.hi/2 and a.hi>=pi.lo/2:sine=I(sine.lo,max(sine.hi,F(1)))
 if a.hi>=pi.lo:cosine=I(min(cosine.lo,F(-1)),cosine.hi);sine=I(min(sine.lo,F(0)),sine.hi)
 return sine,cosine

def material_phi(bank,ray,z,c):

 per bank. Caching that unchanged mathematical factor avoids recalculating
 it; each label's position and full gradient remain independently evaluated.
 """

  cube=(R+roof)**3-3*R*R*z;assert cube>0;r=c['geometry'].interval_cbrt(I(cube),340);tau=I((R**3-cube)/(3*R*R))
  Q=transpose(tuple(c['material_helper'].exp_rotate(tuple(I(int(i==j)) for i in range(3)),scale(A,tau),c) for j in range(3)))
  cache[key]=(cube,r,tau,Q)
