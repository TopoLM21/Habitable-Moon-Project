def enc(x):
    return json.dumps(x,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()
def sha(x):return hashlib.sha256(x).hexdigest()
def rat(x):
    x=F(x);return [x.numerator,x.denominator]
def number(x):return dict(exact_F=rat(x),display_binary64=float(x))
def literal(x):
    assert isinstance(x,(int,float)) and not isinstance(x,bool) and math.isfinite(x)
    return F(x)
def rational(x):
    assert isinstance(x,list) and len(x)==2 and all(isinstance(a,int) for a in x)
    return F(*x)
def close_check(a,b,scale_factor):
    scale=max(abs(a),abs(b),F(sys.float_info.min))
    return dict(residual=number(a-b),allowed_absolute_residual=number(scale_factor*scale),
                within_source_tolerance=abs(a-b)<=scale_factor*scale)
