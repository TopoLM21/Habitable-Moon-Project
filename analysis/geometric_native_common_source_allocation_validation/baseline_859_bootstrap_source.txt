def enc(x):
    return json.dumps(x, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()

def vd(x):
    return hashlib.sha256(enc(x)).hexdigest()

def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()

def read(path):
    return json.loads(Path(path).read_bytes())

def sealed(path, key):
    raw = read(path)
    body = dict(raw)
    assert body.pop(key) == vd(body), str(path)
    return raw

def load(path, name, sha):
    assert digest(path) == sha, str(path)
    m = types.ModuleType(name)
    m.__file__ = str(path)
    sys.modules[name] = m
    exec(compile(Path(path).read_bytes(), str(path), 'exec'), m.__dict__)
    return m
