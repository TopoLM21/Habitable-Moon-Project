def encode(value):
    return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()


def sha(path):
    h=hashlib.sha256()
    with open(path,'rb') as f:
        while chunk:=f.read(1<<20): h.update(chunk)
    return h.hexdigest()


def fraction(raw):
    if type(raw) is int: return F(raw)
    if type(raw) is not list or len(raw)!=2 or any(type(v) is not int for v in raw):
        raise ValueError('Exact scalar requires literal rational pair')
    if raw[1]<=0: raise ValueError('Rational denominator must be positive')
    return F(*raw)


class Interval:
    __slots__=('lo','hi')
    def __init__(self,lo,hi=None):
        self.lo=F(lo);self.hi=self.lo if hi is None else F(hi)
        if self.lo>self.hi: raise ValueError('Reversed interval')
    def other(self,x): return x if isinstance(x,Interval) else Interval(x)
    def __add__(self,x):
        x=self.other(x);return Interval(self.lo+x.lo,self.hi+x.hi)
    __radd__=__add__
    def __neg__(self): return Interval(-self.hi,-self.lo)
    def __sub__(self,x): return self+-self.other(x)
    def __rsub__(self,x): return self.other(x)+-self
    def __mul__(self,x):
        x=self.other(x);cs=(self.lo*x.lo,self.lo*x.hi,self.hi*x.lo,self.hi*x.hi)
        return Interval(min(cs),max(cs))
    __rmul__=__mul__
    def __truediv__(self,x):
        x=self.other(x)
        if x.lo<=0<=x.hi: raise ZeroDivisionError('Interval divisor contains zero')
        return self*Interval(1/x.hi,1/x.lo)
    def center(self): return (self.lo+self.hi)/2


def interval(raw):
    if type(raw) is dict and set(raw)=={'lo','hi'}:
        return Interval(fraction(raw['lo']),fraction(raw['hi']))
    return Interval(fraction(raw))


def center(raw): return interval(raw).center()


def nested(raw,fn):
    if isinstance(raw,dict) and set(raw)=={'lo','hi'}: return fn(raw)
    if isinstance(raw,list) and len(raw)==2 and all(type(x) is int for x in raw): return fn(raw)
    if isinstance(raw,list): return tuple(nested(x,fn) for x in raw)
    return fn(raw)


class Polynomial2:
    """Exact truncated multivariate polynomial in monomials of degree<=2.

    Diagonal quadratic coefficients are Hessian/2; off diagonal coefficients
    are the full Hessian entry. Multiplication is ordinary polynomial
    convolution, independent of the production variational formula.
    """
    __slots__=('n','coefficients')
    def __init__(self,n,coefficients):
        self.n=n;self.coefficients={k:F(v) for k,v in coefficients.items() if v}
    @classmethod
    def jet(cls,value,gradient,hessian):
        n=len(gradient)
        coefficients={():F(value)}
        for i,x in enumerate(gradient): coefficients[(i,)]=F(x)
        for i in range(n):
            for j in range(i,n): coefficients[(i,j)]=F(hessian[i][j])/(2 if i==j else 1)
        return cls(n,coefficients)
    def other(self,x):
        return x if isinstance(x,Polynomial2) else Polynomial2(self.n,{():F(x)})
    def __add__(self,x):
        x=self.other(x)
        if x.n!=self.n: raise ValueError('Polynomial dimensions differ')
        out=dict(self.coefficients)
        for k,v in x.coefficients.items(): out[k]=out.get(k,F(0))+v
        return Polynomial2(self.n,out)
    __radd__=__add__
    def __neg__(self): return Polynomial2(self.n,{k:-v for k,v in self.coefficients.items()})
    def __sub__(self,x): return self+-self.other(x)
    def __rsub__(self,x): return self.other(x)+-self
    def __mul__(self,x):
        x=self.other(x);out={}
        left0=self.coefficients.get((),F(0));right0=x.coefficients.get((),F(0))
        for k,v in self.coefficients.items(): out[k]=v*right0
        for k,v in x.coefficients.items():
            if k: out[k]=out.get(k,F(0))+left0*v
        left1=[(k[0],v) for k,v in self.coefficients.items() if len(k)==1]
        right1=[(k[0],v) for k,v in x.coefficients.items() if len(k)==1]
        for i,v in left1:
            for j,w in right1:
                k=(min(i,j),max(i,j));out[k]=out.get(k,F(0))+v*w
        return Polynomial2(self.n,out)
    __rmul__=__mul__
    def __truediv__(self,x):
        return self*F(1,F(x))
    def value(self): return self.coefficients.get((),F(0))
    def gradient(self): return tuple(self.coefficients.get((i,),F(0)) for i in range(self.n))
    def hessian(self):
        return tuple(tuple(self.coefficients.get(tuple(sorted((i,j))),F(0))*(2 if i==j else 1)
            for j in range(self.n)) for i in range(self.n))


def scalar_potential(A,energies,forces,stiffness,tip,remainder,half_norm):
    p=[tip[i]+sum((A[i][j]*remainder[j] for j in range(8)),0) for i in range(8)]
    value=half_norm+sum(energies)
    for i in range(8):
        value=value+forces[i]*p[i]+stiffness[i]*p[i]*p[i]/2
        for j in range(8): value=value-remainder[i]*A[i][j]*remainder[j]/2
    return value

def compare(value,raw,label,counts):
    enclosure=interval(raw)
    if not enclosure.lo<=value<=enclosure.hi:
        raise AssertionError('Independent exact polynomial scalar outside '+label)
    counts['exact_polynomial_scalar_containment_checks']+=1


def compare_operator(poly,response,label,counts):
    compare(poly.value(),response['energy_j'],label+'.energy',counts)
    gradient=poly.gradient();hessian=poly.hessian()
    rg=response.get('gradient',response.get('gradient_n'))
    rh=response.get('hessian',response.get('hessian_n_m'))
    if len(rg)!=poly.n or len(rh)!=poly.n:
        raise AssertionError('Incomplete global derivative arrays')
    for i in range(poly.n):
        compare(gradient[i],rg[i],label+'.G'+str(i),counts)
        if len(rh[i])!=poly.n: raise AssertionError('Incomplete Hessian row')
        for j in range(poly.n):
            compare(hessian[i][j],rh[i][j],label+'.H'+str(i)+','+str(j),counts)


def uni_multiply(first,second):
    out=[F(0)]*(len(first)+len(second)-1)
    for i,x in enumerate(first):
        for j,y in enumerate(second): out[i+j]+=x*y
    return out


def uni_add(first,second):
    out=[F(0)]*max(len(first),len(second))
    for i,x in enumerate(first): out[i]+=x
    for i,x in enumerate(second): out[i]+=x
    return out


def uni_scale(first,scalar): return [F(scalar)*x for x in first]


def uni_value(coefficients,x):
    out=F(0)
    for c in reversed(coefficients): out=out*x+c
    return out


def uni_derivative(coefficients): return [i*x for i,x in enumerate(coefficients)][1:]


def uni_energy(A,energies,forces,stiffness,tip_coefficients,r_coefficients,s_coefficients):
    p=[uni_add(tip_coefficients[i],
        [sum((A[i][j]*r_coefficients[j][k] for j in range(8)),F(0)) for k in range(3)]) for i in range(8)]
    out=uni_add([sum(energies)],s_coefficients)
    for i in range(8):
        out=uni_add(out,uni_scale(p[i],forces[i]))
        out=uni_add(out,uni_scale(uni_multiply(p[i],p[i]),stiffness[i]/2))
        for j in range(8):
            out=uni_add(out,uni_scale(uni_multiply(r_coefficients[i],r_coefficients[j]),-A[i][j]/2))
    return out


def overlap(first,second,label,counts):

        raise AssertionError('Independent exact intervals conflict: '+label)
    counts['independent_source_interval_overlap_checks']+=1

def source_quartic_center(source):
    """Differentiate/recover from original saved source moments, never nativecoeffs."""
    rows={row['label']:row['response'] for row in source['evaluation_queries']}
    e=rows['registered_cold_endpoint'];raw=e['integrals'];banks=source['cold_continuum_energy_system']['banks']
    a=tuple(map(fraction,e['coordinate_increments']))
    directional=lambda gradient:sum((ai*center(x) for ai,x in zip(a,gradient)),F(0))
    second=lambda h:sum((a[i]*a[j]*center(h[i][j]) for i in range(56) for j in range(56)),F(0))
    rvalue=tuple(map(center,raw['projection_remainder_n']))
    rderivative=tuple(directional(row) for row in raw['projection_remainder_gradient_n_m'])
    r=[[F(0),2*v-d,d-v] for v,d in zip(rvalue,rderivative)]
    s=center(raw['remainder_strain_inner_product_j'])/2
    ds=directional(raw['remainder_local_gradient_n']);dds=second(raw['remainder_local_hessian_n_m'])
    # Solve interpolation equations by Gaussian elimination. This is a
    # distinct derivation from the production closed coefficient formula.
    equations=[[F(1),F(1),F(1),s],[F(2),F(3),F(4),ds],[F(2),F(6),F(12),dds]]
    for i in range(3):
        pivot=equations[i][i];equations[i]=[x/pivot for x in equations[i]]
        for j in range(3):
            if i==j: continue
            coefficient=equations[j][i]
            equations[j]=[x-coefficient*y for x,y in zip(equations[j],equations[i])]
    scoeff=[F(0),F(0)]+[row[3] for row in equations]
    A=tuple(tuple(map(center,row)) for row in e['gram_inverse_certificate']['inverse_gram_bounds_m_n'])
    t=[[F(0),a[b['tip_coordinate_index']]] for b in banks]
    energies=tuple(fraction(b['source_reference_energy_j']) for b in banks)
    forces=tuple(fraction(b['source_reference_force_n']) for b in banks)
    stiffness=tuple(fraction(b['source_old_tip_stiffness_n_m']) for b in banks)
    coefficients=uni_energy(A,energies,forces,stiffness,t,r,scoeff)
    return coefficients,r,scoeff
