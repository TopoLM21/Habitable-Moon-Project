def digest(path):
 h=hashlib.sha256()
 with Path(path).open('rb') as stream:
  for block in iter(lambda:stream.read(1<<20),b''):h.update(block)

def exp_rotate(v,rotation,c):
 """Independent Cartesian exponential and proven absolute Taylor remainder.

 Each operation is outward-rounded at320 decimal digits.  Infinity-norm
 matrix bound ||[a]x|| <= sum|a| <1/4 gives a geometric tail majorant.
 """
 I=c['I'];a=tuple(rounded(x,c) for x in rotation);v=tuple(rounded(x,c) for x in v)
 alpha=sum((max(abs(x.lo),abs(x.hi)) for x in a),F(0));assert alpha<F(1,4)
 term=v;total=v;N=220
 for n in range(1,N+1):
  term=tuple(rounded(x/n,c) for x in cross(a,term));total=tuple(rounded(x+y,c) for x,y in zip(total,term))
 mag=max(max(abs(x.lo),abs(x.hi)) for x in v)
 tail=mag*alpha**(N+1)/math.factorial(N+1)/(1-alpha/F(N+2))
 return tuple(rounded(x+I(-tail,tail),c) for x in total)
