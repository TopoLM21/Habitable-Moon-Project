def enc(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False, allow_nan=False).encode()


def vd(value):
    return hashlib.sha256(enc(value)).hexdigest()


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1 << 20), b''):
            h.update(block)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def sealed(path, key):
    raw = read(path)
    body = dict(raw)
    assert body.pop(key) == vd(body), str(path)
    return raw


def load(path, name, pin):
    assert digest(path) == pin, str(path)
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module
    exec(compile(Path(path).read_bytes(), str(path), 'exec'), module.__dict__)
    return module


def source_map():
    return {str(p.relative_to(ROOT)): digest(p)
            for folder in ('tectonics', 'tests', 'scripts', 'analysis')
            for p in sorted((ROOT / folder).rglob('*.py'))}

    assert manifest['closed_release_output_count'] == 4 and manifest['fresh_tests'] == 18
    assert len(manifest['files']) == 24
    assert manifest['wrapper_selfseal'] == WRAPPER_SEAL
    assert manifest['independent_audit_selfseal'] == AUDIT_SEAL
    assert manifest['physical_candidate_status'] == SCOPE
    assert manifest['original_point_1_completed'] is False
    assert manifest['accepted_original_native_intervals'] == 1 and manifest['accepted_new_native_intervals'] == 0
    assert manifest['original_point_1_remaining_work_blocks'] == 1 and len(manifest['remaining_physical_conditions']) == 3
    for name, row in manifest['files'].items():
        path = ROOT / name
        assert path.resolve().is_relative_to(ROOT) and str(path.relative_to(ROOT)) == name, name
        assert path.stat().st_size == row['bytes'] and digest(path) == row['sha256'], name

def run(*, output_path=OUT):
    result_path = Path(output_path)
    assert result_path.resolve().is_relative_to(Path('/tmp')) and result_path.suffix == '.json'
    assert result_path.resolve() != PRIOR_OUT.resolve(), 'Use a separate scratch result path'
    report = verify_867()
    report['audit_sha256'] = vd(report)
    result_path.write_bytes(enc(report) + b'\n')
    print(enc(report).decode())
    return report


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--out', default=str(OUT))
    args = parser.parse_args()
    run(output_path=args.out)
