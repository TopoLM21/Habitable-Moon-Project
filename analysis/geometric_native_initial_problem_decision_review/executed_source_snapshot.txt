def enc(value):
    return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()

def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as stream:
        while block:=stream.read(1<<20):h.update(block)
    return h.hexdigest()

def sealed(path):
    value=json.loads(Path(path).read_bytes())
    key='report_sha256' if 'report_sha256' in value else 'sha256'
    assert value[key]==hashlib.sha256(enc({k:v for k,v in value.items() if k!=key})).hexdigest()
    return value

def add(*polynomials):
    result={}
    for polynomial in polynomials:
        for exponent,coefficient in polynomial.items():
            result[exponent]=result.get(exponent,F(0))+coefficient
    return {e:c for e,c in result.items() if c}

def mul(left,right):
    result={}
    for e,c in left.items():
        for f,d in right.items():
            exponent=tuple(a+b for a,b in zip(e,f))
            result[exponent]=result.get(exponent,F(0))+c*d
    return {e:c for e,c in result.items() if c}

def derivative(polynomial,axis):
    result={}
    for exponent,coefficient in polynomial.items():
        if exponent[axis]:
            target=list(exponent);target[axis]-=1
            result[tuple(target)]=coefficient*exponent[axis]
    return result

def negate(polynomial):return {e:-c for e,c in polynomial.items()}
def at(polynomial,point):
    return sum((c*point[0]**e[0]*point[1]**e[1]*point[2]**e[2] for e,c in polynomial.items()),F(0))
def integral_cube(polynomial):
    return sum((c*F(8,(e[0]+1)*(e[1]+1)*(e[2]+1))
                for e,c in polynomial.items() if all(n%2==0 for n in e)),F(0))
def pair(value):return [value.numerator,value.denominator]

def proof():
    # b(s)=(1-s^2)^4 inside [-1,1], zero outside. b and derivatives
    # through order3 vanish at both boundaries; its compact extension is C3.
    b=(F(1),F(0),F(-4),F(0),F(6),F(0),F(-4),F(0),F(1))
    one_d=list(b);boundary=[]
    for order in range(4):
        values=[sum((c*F(x)**i for i,c in enumerate(one_d)),F(0)) for x in (-1,1)]
        assert values==[F(0),F(0)]
        boundary.append([pair(v) for v in values])
        one_d=[F(i)*one_d[i] for i in range(1,len(one_d))]
    phi={(i,j,k):a*c*d for i,a in enumerate(b) for j,c in enumerate(b)
                        for k,d in enumerate(b) if a*c*d}
    hessian=[[derivative(derivative(phi,i),j) for j in range(3)] for i in range(3)]
    laplacian=add(*(hessian[i][i] for i in range(3)))
    stress=[[add(laplacian if i==j else {},negate(hessian[i][j])) for j in range(3)] for i in range(3)]
    divergence=[add(*(derivative(stress[i][j],j) for j in range(3))) for i in range(3)]
    assert divergence==[{}, {}, {}]
    center=[[at(stress[i][j],(F(0),)*3) for j in range(3)] for i in range(3)]
    assert center==[[F(-16) if i==j else F(0) for j in range(3)] for i in range(3)]
    norm_squared=sum((integral_cube(mul(row,row)) for line in stress for row in line),F(0))
    assert norm_squared>0
    return dict(formula='delta_sigma = I*Laplacian(phi) - Hessian(phi)',
        potential='phi=b(x)b(y)b(z), b(s)=(1-s^2)^4 for |s|<1; b=0 otherwise',
        polynomial_inside_support_only=True,compact_extension_is_C3=True,
        potential_boundary_derivatives_order0_to3_exact=boundary,
        all_three_divergence_polynomials_exact_zero=True,
        nonzero_center_stress_matrix=[[pair(v) for v in row] for row in center],
        positive_exact_stress_L2_norm_squared_on_unit_cube=pair(norm_squared),
        construction_can_be_scaled_and_placed_strictly_inside_an_open_parent=True,
        stress_and_traction_vanish_outside_support=True,
        example_is_not_a_recovered_historical_stress_field=True,
        example_is_not_a_nonuniqueness_claim_for_a_fully_specified_elastic_BVP=True,
        energy_caveat='Finite force observations do not determine the field. A full elastic energy could select it only after geometry, constitutive law and boundary/history data are specified; saved reduced energies are not that missing full-volume measurement.')
